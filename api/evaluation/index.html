    API Reference — evolzen.evaluation

    Procedural generator for held-out novel mini-environments (Phase 2.4).

    Never reuse a small, fixed set of held-out environments across evaluation rounds — this is a direct, actionable lesson from the field's own ARC-AGI-3 public/private gap (a harness went from 30.2% cold to 96.4% purely through repeated exposure to the same 25 games). Every evaluation round must use genuinely fresh instances.

    Citations: - Cobbe et al. (Procgen, ICML 2020, DOI:10.48550/arXiv.1912.01588) for seed-stratified procedural generation; - Chollet (On the Measure of Intelligence, arXiv:1911.01547; ARC-AGI-2, arXiv:2505.11831).

    Empirical fresh-world adaptation benchmarks remain untested until Phase 2.4 experiments are executed on target hardware (AGENTS.md Rules #4 and #5).

    class ProceduralEnvironmentGenerator:
    """Generates a fresh, never-before-seen environment instance within the Phase 1.1 domain
    family, for each evaluation round."""

    def __init__(self, seed: int = 42) -> None:
        self._seed = seed
        self._round = 0
        self._generated_configs: list[dict[str, Any]] = []

    @property
    def round_count(self) -> int:
        return self._round

    @property
    def generated_configs(self) -> list[dict[str, Any]]:
        return list(self._generated_configs)

    def next_environment(self) -> SocialSandboxEngine:
        """Generate the next fresh, non-reused environment instance.

        Guarantees:
        (a) genuinely novel parameters each call (seed, locations, agent counts);
        (b) novel parameter tracking so the evaluation report can prove zero reuse.
        """
        self._round += 1
        digest = hashlib.sha256(f"{self._seed}:round:{self._round}".encode()).hexdigest()
        round_seed = int(digest[:8], 16)

        # Procedural variation across topologies and capacities
        num_locations = 2 + (round_seed % 5)  # 2..6
        num_agents = 2 + ((round_seed // 5) % 5)  # 2..6
        max_ticks = 200 + (round_seed % 800)  # 200..999

        config = {
            "round": self._round,
            "seed": round_seed,
            "num_locations": num_locations,
            "num_agents": num_agents,
            "max_ticks": max_ticks,
        }

        # Enforce strict non-repetition across seeds
        if any(c["seed"] == round_seed for c in self._generated_configs):
            raise RuntimeError(f"Seed collision detected in procedural environment generation: {round_seed}")

        self._generated_configs.append(config)

        return SocialSandboxEngine(
            default_seed=round_seed,
            num_locations=num_locations,
            num_agents=num_agents,
            max_ticks=max_ticks,
        )

    def has_reused_instances(self) -> bool:
        """Verify that all generated environment seeds are strictly unique."""
        seen_seeds = {c["seed"] for c in self._generated_configs}
        return len(seen_seeds) < len(self._generated_configs)

    def generate_suite(self, num_instances: int) -> list[SocialSandboxEngine]:
        """Generate a batch of fresh, un-reused environment instances."""
        if num_instances <= 0:
            return []
        return [self.next_environment() for _ in range(num_instances)]

    RHAE-style (Relative Human Action Efficiency) scoring (Phase 2.4).

    Modeled on ARC-AGI-3's own metric: penalizes wasted exploratory actions, not just eventual correctness, since the field's benchmark treats how much interaction it takes to learn something new as the actual measure of intelligence, not raw eventual accuracy.

    @dataclass(frozen=True)
class RHAEResult:
    system_actions: int
    human_reference_actions: int
    solved: bool

    @property
    def score(self) -> float:
        """RHAE score in [0, 1+]. 1.0 means the system matched the human reference exactly;
        > 1.0 means the system was *more* action-efficient than the human reference."""
        if not self.solved or self.system_actions == 0:
            return 0.0
        return self.human_reference_actions / self.system_actions

    class RHAEScorer:
    """Aggregates RHAEResults across an evaluation round into a single reportable score."""

    def __init__(self) -> None:
        self._results: list[RHAEResult] = []

    def record(self, result: RHAEResult) -> None:
        self._results.append(result)

    def mean_rhae(self) -> float:
        if not self._results:
            return 0.0
        return sum(r.score for r in self._results) / len(self._results)

    def solve_rate(self) -> float:
        if not self._results:
            return 0.0
        return sum(1 for r in self._results if r.solved) / len(self._results)

    Final honest evaluation report (Phase 2.5).

    Every claim in the report must be traceable to a specific measured result. Critically, the report must explicitly measure and disclose the gap between reused and genuinely fresh held-out environments for this specific system — not a footnote, a citable finding in its own right.

    @dataclass
class EvaluationReport:
    """Structured container for the Phase 2.5 report.

    Deliberately has no "headline claim" field with free text — every reported number must
    come from one of the typed fields below, each traceable to a specific phase's success
    metric, so the report cannot accidentally drift into an unsupported claim.
    """

    in_sandbox_metrics: dict[str, float] = field(default_factory=dict)  # Phases 1.1-2.1
    containment_verified: bool = False  # Phase 2.2
    recognition_ablation_results: dict[str, float] = field(default_factory=dict)  # Phase 2.3
    held_out_rhae_score: float | None = None  # Phase 2.4
    human_reference_rhae: float | None = None  # Phase 2.4 calibration
    frontier_model_reference_rhae: float | None = None  # Phase 2.4 calibration
    reused_vs_fresh_score_gap: float | None = None  # Phase 2.5's most important honesty check
    negative_findings: list[str] = field(default_factory=list)

    def summary(self) -> str:
        """Render a plain-text summary. Raises if the report is incomplete, on purpose — an
        incomplete report should never be silently rendered as if it were final."""
        if self.held_out_rhae_score is None:
            raise ValueError("held_out_rhae_score is required before a report can be summarized.")
        if self.reused_vs_fresh_score_gap is None:
            raise ValueError(
                "reused_vs_fresh_score_gap is required before a report can be summarized "
                "(see the Phase 2.5 honesty requirement)."
            )
        lines = [
            f"Held-out RHAE score: {self.held_out_rhae_score:.3f}",
            f"Human reference: {self.human_reference_rhae}",
            f"Frontier-model reference: {self.frontier_model_reference_rhae}",
            f"Reused-vs-fresh score gap: {self.reused_vs_fresh_score_gap:.3f}",
            f"Containment verified: {self.containment_verified}",
        ]
        if self.negative_findings:
            lines.append("Negative findings:")
            lines.extend(f"  - {f}" for f in self.negative_findings)
        return "\n".join(lines)
