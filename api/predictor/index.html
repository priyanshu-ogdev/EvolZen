    API Reference — evolzen.predictor

    PPO/GRPO-style grounded predictor and planner (Phase 2.1).

    Reward = accuracy of predicted agent intent/outcome against the environment's real, verifiable result. Deliberately no pre-committed accuracy target — reporting whatever accuracy is actually measured on target hardware (AGENTS.md Rules #4 and #5).

    Cites: - DeepSeekMath / GRPO (Shao et al., arXiv:2402.03300) for critic-free group-relative policy optimization; - CLEAR (Rolnick et al., NeurIPS 2019, arXiv:1811.11682) for mixed on-policy / off-policy rehearsal; - Wang et al. (2024, IEEE TPAMI, DOI:10.1109/TPAMI.2024.3367329) for continual learning transfer metrics; - CLaaS (arXiv:2606.05559) for continual learning as an infrastructure service.

    class Planner:
    """Trains via policy-gradient RL (GRPO) against verifiable environment outcomes."""

    def __init__(
        self,
        rehearsal_buffer: RehearsalBuffer,
        dynamics: WorldDynamicsPredictor | None = None,
        *,
        lr: float = 0.05,
        grpo_clip_eps: float = 0.2,
        grpo_beta_kl: float = 0.01,
    ) -> None:
        self._rehearsal_buffer = rehearsal_buffer
        self._dynamics = dynamics if dynamics is not None else WorldDynamicsPredictor()
        self._lr = lr
        self._grpo_clip_eps = grpo_clip_eps
        self._grpo_beta_kl = grpo_beta_kl
        # Parametric action logits for GRPO policy
        self._policy_logits: dict[str, float] = defaultdict(float)
        self._update_count: int = 0
        self._metrics_tracker = ContinualMetricsTracker()

    @property
    def dynamics(self) -> WorldDynamicsPredictor:
        return self._dynamics

    @property
    def rehearsal_buffer(self) -> RehearsalBuffer:
        return self._rehearsal_buffer

    @property
    def metrics_tracker(self) -> ContinualMetricsTracker:
        return self._metrics_tracker

    @property
    def update_count(self) -> int:
        return self._update_count

    def policy_probabilities(self, candidate_actions: Sequence[str]) -> dict[str, float]:
        """Compute softmax policy probabilities over candidate action types."""
        if not candidate_actions:
            return {}
        logits = [self._policy_logits[a] for a in candidate_actions]
        max_l = max(logits)
        exp_logits = [math.exp(val - max_l) for val in logits]
        total_exp = sum(exp_logits)
        if total_exp <= 0.0:
            prob = 1.0 / len(candidate_actions)
            return {a: prob for a in candidate_actions}
        return {a: e / total_exp for a, e in zip(candidate_actions, exp_logits, strict=False)}

    def predict(self, transition_context: list[Transition]) -> PredictionResult:
        """Predict the next agent intent/outcome given recent context.

        Loads current transition dynamics and policy preference, scored against the
        real `Transition.emission` that actually occurs in the environment.
        """
        if not transition_context:
            return PredictionResult(
                predicted_emission={"accepted": True, "reason": "default_context", "rewards": [0.0] * 8},
                confidence=0.1,
                reward_decomposition={dim: 0.0 for dim in REWARD_DIMENSIONS},
            )

        last_t = transition_context[-1]
        actor_id = last_t.action.agent_id
        current_state = last_t.next_state

        # Known candidate action types
        known_actions = list(self._policy_logits.keys())
        if not known_actions:
            known_actions = ["wait", "gather", "move", "craft", "trade", "share"]

        probs = self.policy_probabilities(known_actions)
        best_action_type = max(probs.items(), key=lambda kv: kv[1])[0] if probs else "wait"

        candidate_action = Action(agent_id=actor_id, action_type=best_action_type)
        pred_emission, pred_next_state = self._dynamics.predict(
            state=current_state,
            action=candidate_action,
            seed=last_t.seed,
        )

        pred_rewards = pred_emission.data.get("rewards", [0.0] * 8)
        reward_decomp = {
            name: float(pred_rewards[i]) if i < len(pred_rewards) else 0.0
            for i, name in enumerate(REWARD_DIMENSIONS)
        }

        confidence = float(pred_emission.data.get("confidence", 0.5))

        return PredictionResult(
            predicted_emission=pred_emission.data,
            confidence=confidence,
            predicted_next_state=pred_next_state,
            action=candidate_action,
            reward_decomposition=reward_decomp,
            emission=pred_emission,
        )

    def plan_action(
        self,
        state: State,
        agent_id: str,
        candidate_actions: list[Action],
    ) -> tuple[Action, PredictionResult]:
        """Evaluate candidate actions and select the action optimizing predicted outcome."""
        if not candidate_actions:
            fallback_action = Action(agent_id=agent_id, action_type="wait")
            res = PredictionResult(
                predicted_emission={"accepted": True, "reason": "wait", "rewards": [0.0] * 8},
                confidence=0.5,
            )
            return fallback_action, res

        action_types = [a.action_type for a in candidate_actions]
        probs = self.policy_probabilities(action_types)

        best_score = -float("inf")
        best_action = candidate_actions[0]
        best_result = PredictionResult(
            predicted_emission={},
            confidence=0.0,
        )

        for act in candidate_actions:
            pred_emission, pred_next = self._dynamics.predict(state=state, action=act)
            rewards = pred_emission.data.get("rewards", [0.0] * 8)
            reward_sum = sum(float(r) for r in rewards)
            policy_bias = probs.get(act.action_type, 0.0)

            # Combined score = predicted reward + policy prior
            composite_score = reward_sum + 0.1 * policy_bias
            if composite_score > best_score:
                best_score = composite_score
                best_action = act
                decomp = {
                    name: float(rewards[i]) if i < len(rewards) else 0.0
                    for i, name in enumerate(REWARD_DIMENSIONS)
                }
                best_result = PredictionResult(
                    predicted_emission=pred_emission.data,
                    confidence=float(pred_emission.data.get("confidence", 0.5)),
                    predicted_next_state=pred_next,
                    action=act,
                    reward_decomposition=decomp,
                    emission=pred_emission,
                )

        return best_action, best_result

    def train_step(
        self,
        batch: list[Transition],
        replay_ratio: float = 0.5,
    ) -> dict[str, float]:
        """One GRPO update with CLEAR mixed rehearsal against catastrophic forgetting.

        Cites DeepSeekMath (Shao et al., arXiv:2402.03300) for critic-free group-relative
        policy optimization and Rolnick et al. (CLEAR, NeurIPS 2019) for replay mixing.
        """
        if not batch and len(self._rehearsal_buffer) == 0:
            return {"loss": 0.0, "sample_count": 0.0}

        # CLEAR mixed on-policy / off-policy batching
        total_batch_size = max(len(batch), 1)
        mixed_batch = self._rehearsal_buffer.sample_clear_batch(
            new_transitions=batch,
            total_batch_size=total_batch_size,
            replay_ratio=replay_ratio,
        )
        if not mixed_batch:
            mixed_batch = list(batch)

        # 1. Update supervised world dynamics baseline
        dyn_metrics = self._dynamics.train_batch(mixed_batch)

        # 2. Extract empirical outcomes / verifiable rewards
        rewards: list[float] = []
        action_types: list[str] = []
        for t in mixed_batch:
            r_raw = t.emission.data.get("rewards", [0.0] * 8)
            r_val = sum(float(x) for x in r_raw) if isinstance(r_raw, (list, tuple)) else 0.0
            rewards.append(r_val)
            action_types.append(t.action.action_type)

        # 3. GRPO Group-Relative Advantage Normalization
        mean_r = float(np.mean(rewards)) if rewards else 0.0
        std_r = float(np.std(rewards)) if len(rewards) > 1 else 1.0
        if std_r < 1e-6:
            std_r = 1.0

        advantages = [(r - mean_r) / std_r for r in rewards]

        # 4. Policy gradient update on action logits
        grad_norms: list[float] = []
        for act_type, adv in zip(action_types, advantages, strict=False):
            clipped_adv = max(-self._grpo_clip_eps, min(self._grpo_clip_eps, adv))
            delta = self._lr * clipped_adv
            # KL regularization towards 0.0 (uniform prior)
            self._policy_logits[act_type] = (
                (1.0 - self._grpo_beta_kl) * self._policy_logits[act_type] + delta
            )
            grad_norms.append(abs(delta))

        # 5. Populate rehearsal buffer with the new transitions
        for t in batch:
            self._rehearsal_buffer.add(t)

        self._update_count += 1

        # Calculate policy entropy for monitoring
        distinct_actions = list(self._policy_logits.keys())
        probs = self.policy_probabilities(distinct_actions)
        entropy = -sum(p * math.log(max(1e-9, p)) for p in probs.values()) if probs else 0.0

        return {
            "loss": dyn_metrics.get("loss", 0.0),
            "grpo_advantage_mean": mean_r,
            "grpo_advantage_std": std_r,
            "policy_entropy": entropy,
            "avg_gradient_step": float(np.mean(grad_norms)) if grad_norms else 0.0,
            "sample_count": float(len(mixed_batch)),
            "rehearsal_buffer_size": float(len(self._rehearsal_buffer)),
            "update_count": float(self._update_count),
        }

    @dataclass
class PredictionResult:
    """Structured prediction outcome from the grounded predictor/planner."""

    predicted_emission: dict[str, Any]
    confidence: float
    predicted_next_state: State | None = None
    action: Action | None = None
    reward_decomposition: dict[str, float] = field(default_factory=dict)
    emission: Emission | None = None

    def __post_init__(self) -> None:
        if self.emission is None and isinstance(self.predicted_emission, dict):
            self.emission = Emission(data=self.predicted_emission)

    Deployment-experience rehearsal buffer and continual-learning metrics (Phase 2.1).

    Following CLaaS (arXiv:2606.05559) and CLEAR (Rolnick et al., NeurIPS 2019, arXiv:1811.11682): mixing replayed real (situation -> outcome) pairs from actual deployment into every policy-gradient update is the default continual-learning safeguard for the planner. Continual learning metrics formalize backward transfer, forward transfer, and catastrophic forgetting following Wang et al. (2024, IEEE TPAMI, DOI:10.1109/TPAMI.2024.3367329) and Kirkpatrick et al. (PNAS 2017, DOI:10.1073/pnas.1611835114).

    class ContinualMetricsTracker:
    """Tracks longitudinal transfer, backward transfer, and catastrophic forgetting.

    Implements formal evaluation protocols from Wang et al. (IEEE TPAMI 2024,
    DOI:10.1109/TPAMI.2024.3367329).

    Matrix R[i, j] denotes the performance on task/context j after training on task i.
    Tasks are 1-indexed integers (1, 2, ..., T).
    """

    def __init__(self) -> None:
        # R[train_task][eval_task] -> float
        self._matrix: dict[int, dict[int, float]] = {}
        # Initial zero-shot baselines: baseline[eval_task] -> float
        self._baselines: dict[int, float] = {}

    def set_baseline(self, task_idx: int, score: float) -> None:
        """Set zero-shot performance on a task prior to any sequential training."""
        self._baselines[task_idx] = score

    def record_eval(self, train_task: int, eval_task: int, score: float) -> None:
        """Record the performance on eval_task after completing train_task."""
        if train_task not in self._matrix:
            self._matrix[train_task] = {}
        self._matrix[train_task][eval_task] = score

    def get_score(self, train_task: int, eval_task: int) -> float | None:
        """Retrieve R[train_task, eval_task] if recorded."""
        return self._matrix.get(train_task, {}).get(eval_task)

    def backward_transfer(self, up_to_task: int | None = None) -> float:
        """Compute Backward Transfer (BWT).

        BWT = 1 / (T - 1) * sum_{i=1}^{T-1} (R_{T, i} - R_{i, i})
        Negative values indicate catastrophic forgetting; >= 0 indicates retention or improvement.
        """
        tasks = sorted(self._matrix.keys())
        if len(tasks) < 2:
            return 0.0

        current_t = tasks[-1] if up_to_task is None else up_to_task
        eval_tasks = [t for t in tasks if t < current_t]
        if not eval_tasks:
            return 0.0

        diffs: list[float] = []
        for i in eval_tasks:
            r_t_i = self.get_score(current_t, i)
            r_i_i = self.get_score(i, i)
            if r_t_i is not None and r_i_i is not None:
                diffs.append(r_t_i - r_i_i)

        return sum(diffs) / len(diffs) if diffs else 0.0

    def forward_transfer(
        self,
        baseline_scores: dict[int, float] | None = None,
        up_to_task: int | None = None,
    ) -> float:
        """Compute Forward Transfer (FWT).

        FWT = 1 / (T - 1) * sum_{i=2}^{T} (R_{i-1, i} - b_i)
        where b_i is zero-shot baseline performance on task i.
        """
        baselines = baseline_scores if baseline_scores is not None else self._baselines
        tasks = sorted(self._matrix.keys())
        if len(tasks) < 2:
            return 0.0

        max_t = tasks[-1] if up_to_task is None else up_to_task
        target_tasks = [t for t in tasks if t >= 2 and t <= max_t]
        if not target_tasks:
            return 0.0

        diffs: list[float] = []
        for i in target_tasks:
            prev_t = i - 1
            r_prev_i = self.get_score(prev_t, i)
            b_i = baselines.get(i, 0.0)
            if r_prev_i is not None:
                diffs.append(r_prev_i - b_i)

        return sum(diffs) / len(diffs) if diffs else 0.0

    def forgetting(self, eval_task: int, current_task: int | None = None) -> float:
        """Compute catastrophic forgetting on a specific task j.

        F_j = max_{k in {1, ..., T-1}} R_{k, j} - R_{T, j}
        """
        tasks = sorted(self._matrix.keys())
        if not tasks:
            return 0.0

        current_t = tasks[-1] if current_task is None else current_task
        r_t_j = self.get_score(current_t, eval_task)
        if r_t_j is None:
            return 0.0

        prior_scores = [
            self.get_score(k, eval_task)
            for k in tasks
            if k < current_t and self.get_score(k, eval_task) is not None
        ]
        valid_priors = [s for s in prior_scores if s is not None]
        if not valid_priors:
            return 0.0

        return max(0.0, max(valid_priors) - r_t_j)

    def average_forgetting(self, up_to_task: int | None = None) -> float:
        """Compute average catastrophic forgetting across all previously learned tasks."""
        tasks = sorted(self._matrix.keys())
        if len(tasks) < 2:
            return 0.0

        current_t = tasks[-1] if up_to_task is None else up_to_task
        prior_tasks = [t for t in tasks if t < current_t]
        if not prior_tasks:
            return 0.0

        forgetting_vals = [self.forgetting(j, current_t) for j in prior_tasks]
        return sum(forgetting_vals) / len(forgetting_vals)

    def summary(self) -> dict[str, Any]:
        """Return a structured summary of continual learning evaluation metrics."""
        tasks = sorted(self._matrix.keys())
        return {
            "num_tasks": len(tasks),
            "backward_transfer": self.backward_transfer(),
            "forward_transfer": self.forward_transfer(),
            "average_forgetting": self.average_forgetting(),
            "matrix": {f"task_{i}": {f"eval_{j}": val for j, val in row.items()} for i, row in self._matrix.items()},
        }

    class RehearsalBuffer:
    """Fixed-capacity FIFO buffer of real transitions, sampled during training updates."""

    def __init__(self, capacity: int) -> None:
        if capacity <= 0:
            raise ValueError("capacity must be positive")
        self._capacity = capacity
        self._buffer: deque[Transition] = deque(maxlen=capacity)

    @property
    def capacity(self) -> int:
        return self._capacity

    def add(self, transition: Transition) -> None:
        """Add a single verified transition to the rehearsal buffer."""
        self._buffer.append(transition)

    def extend(self, transitions: Sequence[Transition]) -> None:
        """Add multiple verified transitions to the rehearsal buffer."""
        for t in transitions:
            self._buffer.append(t)

    def clear(self) -> None:
        """Clear all transitions from the rehearsal buffer."""
        self._buffer.clear()

    def sample(self, batch_size: int, rng: random.Random | None = None) -> list[Transition]:
        """Sample uniformly at random without replacement from the buffer."""
        if batch_size > len(self._buffer):
            raise ValueError(
                f"Requested batch_size={batch_size} exceeds buffer size={len(self._buffer)}"
            )
        buf_list = list(self._buffer)
        if rng is not None:
            return rng.sample(buf_list, batch_size)
        return random.sample(buf_list, batch_size)

    def load_from_store(
        self,
        store: TransitionLogStore,
        max_records: int | None = None,
        run_id: str | None = None,
        branch_id: str | None = None,
    ) -> int:
        """Stream historical transitions from DuckDB TransitionLogStore into the buffer.

        Returns the number of transitions actually added.
        """
        transitions = store.list_transitions(run_id=run_id, branch_id=branch_id)
        if max_records is not None and max_records > 0:
            transitions = transitions[-max_records:]
        self.extend(transitions)
        return len(transitions)

    def sample_clear_batch(
        self,
        new_transitions: Sequence[Transition],
        total_batch_size: int,
        replay_ratio: float = 0.5,
        rng: random.Random | None = None,
    ) -> list[Transition]:
        """CLEAR mixed on-policy / off-policy batch sampling (Rolnick et al., NeurIPS 2019).

        Mixes incoming on-policy transitions with historical off-policy replay according
        to `replay_ratio` in [0.0, 1.0].
        """
        if total_batch_size <= 0:
            return []
        if not new_transitions and not self._buffer:
            return []

        ratio = max(0.0, min(1.0, replay_ratio))
        n_replay_target = int(total_batch_size * ratio)
        n_new_target = total_batch_size - n_replay_target

        n_replay = min(n_replay_target, len(self._buffer))
        n_new = min(n_new_target, len(new_transitions))

        # Re-balance if one partition has insufficient records
        if n_replay < n_replay_target:
            n_new = min(len(new_transitions), total_batch_size - n_replay)
        elif n_new < n_new_target:
            n_replay = min(len(self._buffer), total_batch_size - n_new)

        sampled_replay: list[Transition] = []
        if n_replay > 0:
            sampled_replay = self.sample(n_replay, rng=rng)

        sampled_new: list[Transition] = []
        if n_new > 0:
            new_list = list(new_transitions)
            if rng is not None:
                sampled_new = rng.sample(new_list, n_new) if n_new <= len(new_list) else new_list
            else:
                sampled_new = random.sample(new_list, n_new) if n_new <= len(new_list) else new_list

        combined = sampled_replay + sampled_new
        if rng is not None:
            rng.shuffle(combined)
        else:
            random.shuffle(combined)
        return combined

    def __len__(self) -> int:
        return len(self._buffer)
