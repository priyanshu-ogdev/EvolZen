    Phase 1.2 — Unified Agent Profile and Identity Schema.

    Defines the persistent social, institutional, and lineage identity contracts for agents in the EvolZen multi-agent population (docs/ARCHITECTURE.md §3.2, docs/phases/1.2-agents.md, ADR 0006, ADR 0008). Replaces the legacy five-coordinate DriveVector with a rich social profile linked to evolzen.schemas.world.WorldAgentState and the eight-coordinate developmental core (evolzen.schemas.development.AgentDevelopmentState).

    Citations: - Park et al. (Generative Agents, arXiv:2304.03442) for agent social identity, memory, and roles; - Cook et al. (Artificial Generational Intelligence, arXiv:2406.00392) for cultural transfer; - Vassiliades et al. (CVT-MAP-Elites, IEEE TEVC 2018, DOI:10.1109/TEVC.2017.2735550) for behavioral descriptor keys in high-dimensional phenotype archives.

    These contracts are an EvolZen specification choice and remain empirically untested until Phase 1.2 population experiments are run.

    @dataclass
class Agent:
    """Unified developmental agent in the EvolZen multi-agent population."""

    profile: AgentProfile
    development: AgentDevelopmentState = field(repr=False)
    backend: InferenceBackend | None = field(default=None, repr=False)

    @classmethod
    def create(
        cls,
        profile: AgentProfile,
        *,
        seed: int,
        tick: int = 0,
        config: DevelopmentConfig | None = None,
        parents: tuple[Agent, ...] = (),
        backend: InferenceBackend | None = None,
    ) -> Agent:
        """Instantiate a newborn agent with an initialized developmental state."""
        dev_parents = tuple(p.development for p in parents)
        dev_state = born(
            agent_id=profile.agent_id,
            seed=seed,
            tick=tick,
            config=config,
            parents=dev_parents,
        )
        return cls(profile=profile, development=dev_state, backend=backend)

    @property
    def agent_id(self) -> str:
        """Authoritative identifier of this agent."""
        return self.profile.agent_id

    @property
    def tick(self) -> int:
        """Current internal developmental clock tick."""
        return self.development.tick

    @property
    def age(self) -> int:
        """Elapsed ticks since birth."""
        return self.development.tick - self.development.birth.born_at

    @property
    def stage(self) -> LifeStage:
        """Current developmental life stage of the agent."""
        return stage_at(self.development, self.tick)

    def advance(self, tick: int) -> None:
        """Advance the agent's developmental lifecycle to `tick`."""
        self.development = advance(self.development, tick)

    def decide(
        self,
        signal: Signal,
        candidates: tuple[Candidate, ...],
        tick: int,
        *,
        world_context: dict[str, Any] | None = None,
        seed: int | None = None,
    ) -> Decision:
        """Select a candidate action proposal satisfying host-mediated decision invariants."""
        # Update internal motive dynamics and obtain decision proposal
        next_dev = decide(self.development, signal, candidates, tick, seed=seed)
        self.development = next_dev
        pending = self.development.pending
        if pending is None:
            raise RuntimeError(f"Agent {self.agent_id!r} failed to produce a pending decision")
        return pending

    def act_with_backend(
        self,
        signal: Signal,
        candidates: tuple[Candidate, ...],
        tick: int,
        world_context: dict[str, Any],
    ) -> Action:
        """Generate a schema-constrained Action via the configured `InferenceBackend`."""
        if self.backend is None:
            raise RuntimeError(f"Agent {self.agent_id!r} has no configured InferenceBackend")
        decision = self.decide(signal, candidates, tick, world_context=world_context)
        return self.backend.generate_action(
            profile=self.profile,
            signal=signal,
            candidates=candidates,
            motive_weights=decision.ratios,
            world_context=world_context,
        )

    def observe(self, outcome: Outcome) -> None:
        """Ingest a host-verified `Outcome` receipt and update learned skill estimates."""
        self.development = observe(self.development, outcome)

    def teach(
        self,
        child: Agent,
        context_key: str,
        skill_key: str,
        tick: int,
        manual_entry: ManualEntry | None = None,
    ) -> None:
        """Transmit parental advice to `child` without mutating personal evidence or world state."""
        child.development = teach(
            self.development,
            child.development,
            context_key,
            skill_key,
            tick,
            manual_entry=manual_entry,
        )

    def teach_manual(
        self,
        child: Agent,
        manual_entry: ManualEntry,
        tick: int,
    ) -> None:
        """Transmit an active manual entry hypothesis to `child` without mutating personal evidence."""
        child.development = teach_manual(
            self.development,
            child.development,
            manual_entry,
            tick,
        )

    Structured-decoding LLM inference backends for agent action generation (Phase 1.2).

    Every agent action must be a schema-valid evolzen.schemas.transition.Action — that's what makes the Phase 1.1 environment verifiable and evolzen.exploration.verifier meaningful. Rather than parsing free-text model output and hoping it matches the schema, this module uses outlines to constrain generation so the model can only produce valid Action JSON.

    Two backends satisfy the same protocol: TransformersBackend (CPU-friendly default) and VLLMBackend (throughput-optimized for a live population on GPU hardware; requires the gpu extra). Callers depend on InferenceBackend, not either concrete class, so switching backends is a config change, not a code change.

    Design authority: docs/phases/1.2-agents.md, ADR 0006, ADR 0008. Citations: - Willard & Louf (outlines, arXiv:2307.09702) for efficient guided generation with regular expressions and context-free grammars; - Park et al. (Generative Agents, arXiv:2304.03442) for agent reflection and social planning.

    All neural inference mechanisms remain empirically untested until Phase 1.2 experiments are run.

    class TransformersBackend:
    """Default, CPU-friendly backend using `transformers` + `outlines` constrained generation."""

    def __init__(
        self,
        model_id: str = "Qwen/Qwen2.5-0.5B-Instruct",
        *,
        device: str = "cpu",
        offline_fallback: bool = True,
        load_weights: bool = False,
    ) -> None:
        self._model_id = model_id
        self._device = device
        self._offline_fallback = offline_fallback
        self._load_weights = load_weights
        self._generator: Any = None

    def _init_generator(self) -> None:
        if self._generator is not None or not self._load_weights:
            return
        try:
            import outlines
            import torch
            from transformers import AutoModelForCausalLM, AutoTokenizer

            tokenizer = AutoTokenizer.from_pretrained(self._model_id)
            model = AutoModelForCausalLM.from_pretrained(
                self._model_id,
                torch_dtype=torch.float32,
                device_map=self._device,
            )
            outlines_model = outlines.models.Transformers(model, tokenizer)  # type: ignore[arg-type]
            generate_mod = getattr(outlines, "generate", None)
            if generate_mod is not None:
                self._generator = generate_mod.json(outlines_model, Action)
        except Exception as exc:
            if not self._offline_fallback:
                raise RuntimeError(
                    f"Failed to initialize outlines generator for {self._model_id}: {exc}"
                ) from exc
            self._generator = None

    def build_prompt(
        self,
        profile: AgentProfile,
        signal: Signal,
        candidates: tuple[Candidate, ...],
        motive_weights: tuple[float, ...],
        world_context: dict[str, Any],
    ) -> str:
        """Format the agent prompt incorporating social identity, motives, and context."""
        legal_ids = [c.action_id for c in candidates if c.permitted]
        payload = {
            "agent_id": profile.agent_id,
            "roles": profile.roles,
            "norms": profile.adopted_norms,
            "institutions": profile.institutional_affiliations,
            "motive_weights": list(motive_weights),
            "signal": list(signal),
            "legal_candidates": legal_ids,
            "context": world_context,
        }
        return (
            f"You are agent {profile.agent_id}. Given your state and motives, "
            f"choose a valid Action from the legal candidates:\n"
            f"{json.dumps(payload, separators=(',', ':'))}\n"
        )

    def generate_action(
        self,
        profile: AgentProfile,
        signal: Signal,
        candidates: tuple[Candidate, ...],
        motive_weights: tuple[float, ...],
        world_context: dict[str, Any],
    ) -> Action:
        """Generate a schema-valid Action constrained by outlines, with deterministic safe fallback."""
        self._init_generator()
        if self._generator is not None:
            prompt = self.build_prompt(profile, signal, candidates, motive_weights, world_context)
            try:
                result: Action = self._generator(prompt)
                if isinstance(result, Action) and any(result.action_type == at.value for at in ActionType):
                    return result
            except Exception:
                pass  # Fall through to deterministic safe fallback

        # Deterministic fallback: choose the highest-priority permitted candidate
        permitted = [c for c in candidates if c.permitted]
        chosen = permitted[0] if permitted else (candidates[0] if candidates else None)
        if chosen is None:
            return Action(agent_id=profile.agent_id, action_type=ActionType.WAIT)
        return action_from_candidate(chosen.action_id, profile.agent_id)

    class CVTArchive:
    """Centroidal Voronoi Tessellation (CVT) MAP-Elites archive for high-dimensional descriptors.

    Uses fixed centroid niches in descriptor space to prevent exponential cell explosion
    (Vassiliades et al., 2018).
    """

    def __init__(
        self,
        centroids: tuple[tuple[float, ...], ...],
    ) -> None:
        if not centroids:
            raise ValueError("CVTArchive requires at least one centroid")
        self._dimension = len(centroids[0])
        if any(len(c) != self._dimension for c in centroids):
            raise ValueError("All centroids must share the same dimensionality")
        self._centroids = centroids
        self._cells: dict[int, QDArchiveEntry] = {}

    @classmethod
    def uniform_hypercube(
        cls,
        num_centroids: int,
        dimensions: int = 5,
        *,
        seed: int = 42,
    ) -> CVTArchive:
        """Generate deterministic pseudo-random centroid niches in [0, 1]^d."""
        import random

        rng = random.Random(seed)
        generated: list[tuple[float, ...]] = []
        for _ in range(num_centroids):
            point = tuple(round(rng.uniform(0.0, 1.0), 4) for _ in range(dimensions))
            generated.append(point)
        return cls(tuple(generated))

    def _nearest_centroid_index(self, descriptor: tuple[float, ...]) -> int:
        if len(descriptor) != self._dimension:
            raise ValueError(
                f"Descriptor dimension {len(descriptor)} != centroid dimension {self._dimension}"
            )
        best_idx = 0
        min_dist = float("inf")
        for idx, centroid in enumerate(self._centroids):
            dist = sum((a - b) ** 2 for a, b in zip(descriptor, centroid, strict=True))
            if dist < min_dist:
                min_dist = dist
                best_idx = idx
        return best_idx

    def consider(self, entry: QDArchiveEntry) -> bool:
        """Insert `entry` into its nearest centroid Voronoi cell if fitness exceeds incumbent."""
        idx = self._nearest_centroid_index(entry.behavioral_descriptor)
        current = self._cells.get(idx)
        if current is None or entry.fitness > current.fitness:
            self._cells[idx] = entry
            return True
        return False

    def coverage(self) -> int:
        """Number of populated Voronoi cells."""
        return len(self._cells)

    def total_niches(self) -> int:
        """Total number of centroid niches in the tessellation."""
        return len(self._centroids)

    def entries(self) -> list[QDArchiveEntry]:
        """Return all occupied entries."""
        return list(self._cells.values())

    def __init__(self, cell_resolution: float = 0.1) -> None:

    def compute_behavioral_descriptor(
    genome: BirthGenome,
    profile: AgentProfile,
    dimension: int = 5,
) -> tuple[float, ...]:
    """Compute normalized [0.0, 1.0]^d behavioral descriptor from agent genome and social profile.

    Measures phenotypic behavior across orthogonal dimensions (risk-tolerance, status-drive,
    social-cooperativeness, norm-adherence, and resource accumulation momentum).
    """
    if dimension < 1:
        raise ValueError("dimension must be at least 1")
    # Coord 0: Greed/Risk bias (normalized to [0, 1])
    d0 = max(0.0, min(1.0, round((genome.equations[1].bias + 1.0) / 2.0, 4)))
    # Coord 1: Pride/Status momentum
    d1 = max(0.0, min(1.0, round(genome.equations[0].momentum, 4)))
    # Coord 2: Lust/Social affinity context sensitivity
    d2 = max(0.0, min(1.0, round(sum(abs(c) for c in genome.equations[2].context) / 8.0, 4)))
    # Coord 3: Institutional norm cooperativeness
    d3 = max(0.0, min(1.0, round(min(1.0, len(profile.adopted_norms) * 0.25 + len(profile.institutional_affiliations) * 0.25), 4)))
    # Coord 4: Gluttony/Resource accumulation momentum
    d4 = max(0.0, min(1.0, round(genome.equations[4].momentum, 4)))
    coords = [d0, d1, d2, d3, d4]
    while len(coords) < dimension:
        coords.append(0.5)
    return tuple(coords[:dimension])

    class PopulationScheduler:
    """Coordinates lifecycle stepping, generational reproduction, and archive admission."""

    def __init__(
        self,
        engine: SocialSandboxEngine,
        *,
        initial_state: State,
        config: DevelopmentConfig | None = None,
        archive: CVTArchive | None = None,
        store: TransitionLogStore | None = None,
        manual_store: ManualStore | None = None,
        run_id: str = "population-run",
        branch_id: str = "main",
    ) -> None:
        self.engine = engine
        self.state = initial_state
        self.config = config or DevelopmentConfig()
        self.archive = archive or CVTArchive.uniform_hypercube(num_centroids=32, dimensions=5)
        self.store = store
        self.manual_store = manual_store
        self.run_id = run_id
        self.branch_id = branch_id

        self.agents: dict[str, Agent] = {}
        self._init_agents_from_world()

    def _init_agents_from_world(self) -> int:
        """Instantiate Agent entities for newly born agents in the world.

        Returns:
            Number of newly registered agents.
        """
        world = WorldState.from_state(self.state)
        new_count = 0
        for agent_id, w_agent in sorted(world.agents.items()):
            if w_agent.alive and agent_id not in self.agents:
                profile = AgentProfile(
                    agent_id=agent_id,
                    lineage_id=w_agent.lineage_id,
                    generation=w_agent.generation,
                    parents=w_agent.parents,
                    roles=(w_agent.role,) if w_agent.role else ("generalist",),
                    institutional_affiliations=w_agent.institution_ids,
                    adopted_norms=w_agent.adopted_norms,
                )
                parent_agents = tuple(
                    self.agents[p_id]
                    for p_id in w_agent.parents
                    if p_id in self.agents
                )
                init_tick = w_agent.born_at if not parent_agents else world.tick
                agent = Agent.create(
                    profile=profile,
                    seed=world.seed + len(self.agents),
                    tick=init_tick,
                    config=self.config,
                    parents=parent_agents,
                )
                desc = compute_behavioral_descriptor(
                    agent.development.birth.genome,
                    profile,
                    dimension=self.archive._dimension,
                )
                agent.profile = profile.model_copy(update={"behavioral_descriptor": desc})
                if world.tick > agent.tick:
                    agent.advance(world.tick)
                self.agents[agent_id] = agent
                new_count += 1
        return new_count

    def step_agent(self, agent_id: str, *, seed: int | None = None) -> tuple[Transition, Outcome] | None:
        """Step one living agent through the observation -> decision -> host-execution loop."""
        agent = self.agents.get(agent_id)
        if agent is None or agent.stage == LifeStage.DEAD:
            return None

        world = WorldState.from_state(self.state)
        w_agent = world.agents.get(agent_id)
        if w_agent is None or not w_agent.alive:
            return None

        features = observe_features(self.state, agent_id)
        candidates = legal_candidates(self.state, agent_id, config=self.config)

        decision = agent.decide(features, candidates, world.tick, seed=seed)
        transition, outcome = execute_decision(self.engine, self.state, decision, seed=seed)

        agent.observe(outcome)
        self.state = transition.next_state
        self._init_agents_from_world()

        if self.store is not None:
            self.store.append(transition, run_id=self.run_id, branch_id=self.branch_id)

        return transition, outcome

    def run_generation(
        self,
        steps: int = 10,
        *,
        seed: int = 42,
    ) -> PopulationSchedulerMetrics:
        """Execute `steps` world cycles across living agents with teaching and reproduction."""
        transitions_count = 0
        births = 0
        deaths = 0
        teachings = 0

        for step_idx in range(steps):
            world = WorldState.from_state(self.state)
            living_ids = [
                a_id for a_id, a in self.agents.items()
                if a.stage != LifeStage.DEAD and world.agents.get(a_id, None) and world.agents[a_id].alive
            ]

            for a_id in living_ids:
                step_res = self.step_agent(a_id, seed=seed + step_idx)
                if step_res is not None:
                    transitions_count += 1

            # Detect newly born agents
            new_births = self._init_agents_from_world()
            births += new_births

            # Facilitate living parent-child teaching among colocated agents
            curr_world = WorldState.from_state(self.state)
            for a_id, agent in list(self.agents.items()):
                if agent.stage in (LifeStage.ADULT, LifeStage.ELDER):
                    w_parent = curr_world.agents.get(a_id)
                    if w_parent is None or not w_parent.alive:
                        continue
                    # Find living children in the same location
                    for c_id, child in self.agents.items():
                        w_child = curr_world.agents.get(c_id)
                        if (
                            a_id in child.profile.parents
                            and child.stage != LifeStage.DEAD
                            and w_child is not None
                            and w_child.alive
                            and w_child.location_id == w_parent.location_id
                        ):
                            # 1. Empirical skill knowledge teaching
                            if bool(agent.development.knowledge):
                                kn = agent.development.knowledge[0]
                                try:
                                    agent.teach(child, kn.context_key, kn.skill_key, curr_world.tick)
                                    teachings += 1
                                except ValueError:
                                    pass

                            # 2. Living manual hypothesis transmission
                            if self.manual_store is not None:
                                active_entries = self.manual_store.active_entries()
                                if active_entries:
                                    try:
                                        agent.teach_manual(child, active_entries[-1], curr_world.tick)
                                        teachings += 1
                                    except ValueError:
                                        pass

            # Archive mature/retiring agents into CVTArchive
            for a_id, agent in self.agents.items():
                if agent.stage in (LifeStage.ADULT, LifeStage.ELDER, LifeStage.DEAD):
                    w_a = curr_world.agents.get(a_id)
                    fitness = float(w_a.status + sum(w_a.inventory.values())) if w_a else 0.0
                    descriptor = agent.profile.behavioral_descriptor
                    if len(descriptor) == self.archive._dimension:
                        entry = QDArchiveEntry(
                            agent_id=a_id,
                            profile=agent.profile,
                            behavioral_descriptor=descriptor,
                            fitness=fitness,
                        )
                        self.archive.consider(entry)

            # Reconcile deaths
            for a_id, agent in self.agents.items():
                w_a = curr_world.agents.get(a_id)
                if (agent.stage == LifeStage.DEAD or (w_a and not w_a.alive)) and a_id in living_ids:
                    deaths += 1

        active_pop = sum(1 for a in self.agents.values() if a.stage != LifeStage.DEAD)
        final_world = WorldState.from_state(self.state)

        return PopulationSchedulerMetrics(
            ticks_elapsed=final_world.tick,
            total_transitions=transitions_count,
            births=births,
            deaths=deaths,
            teachings=teachings,
            archive_coverage=self.archive.coverage(),
            active_population=active_pop,
        )

    self,
    steps: int = 10,
    *,
    seed: int = 42,
) -> PopulationSchedulerMetrics:
    """Execute `steps` world cycles across living agents with teaching and reproduction."""
    transitions_count = 0
    births = 0
    deaths = 0
    teachings = 0

    for step_idx in range(steps):
        world = WorldState.from_state(self.state)
        living_ids = [
            a_id for a_id, a in self.agents.items()
            if a.stage != LifeStage.DEAD and world.agents.get(a_id, None) and world.agents[a_id].alive
        ]

        for a_id in living_ids:
            step_res = self.step_agent(a_id, seed=seed + step_idx)
            if step_res is not None:
                transitions_count += 1

        # Detect newly born agents
        new_births = self._init_agents_from_world()
        births += new_births

        # Facilitate living parent-child teaching among colocated agents
        curr_world = WorldState.from_state(self.state)
        for a_id, agent in list(self.agents.items()):
            if agent.stage in (LifeStage.ADULT, LifeStage.ELDER):
                w_parent = curr_world.agents.get(a_id)
                if w_parent is None or not w_parent.alive:
                    continue
                # Find living children in the same location
                for c_id, child in self.agents.items():
                    w_child = curr_world.agents.get(c_id)
                    if (
                        a_id in child.profile.parents
                        and child.stage != LifeStage.DEAD
                        and w_child is not None
                        and w_child.alive
                        and w_child.location_id == w_parent.location_id
                    ):
                        # 1. Empirical skill knowledge teaching
                        if bool(agent.development.knowledge):
                            kn = agent.development.knowledge[0]
                            try:
                                agent.teach(child, kn.context_key, kn.skill_key, curr_world.tick)
                                teachings += 1
                            except ValueError:
                                pass

                        # 2. Living manual hypothesis transmission
                        if self.manual_store is not None:
                            active_entries = self.manual_store.active_entries()
                            if active_entries:
                                try:
                                    agent.teach_manual(child, active_entries[-1], curr_world.tick)
                                    teachings += 1
                                except ValueError:
                                    pass

        # Archive mature/retiring agents into CVTArchive
        for a_id, agent in self.agents.items():
            if agent.stage in (LifeStage.ADULT, LifeStage.ELDER, LifeStage.DEAD):
                w_a = curr_world.agents.get(a_id)
                fitness = float(w_a.status + sum(w_a.inventory.values())) if w_a else 0.0
                descriptor = agent.profile.behavioral_descriptor
                if len(descriptor) == self.archive._dimension:
                    entry = QDArchiveEntry(
                        agent_id=a_id,
                        profile=agent.profile,
                        behavioral_descriptor=descriptor,
                        fitness=fitness,
                    )
                    self.archive.consider(entry)

        # Reconcile deaths
        for a_id, agent in self.agents.items():
            w_a = curr_world.agents.get(a_id)
            if (agent.stage == LifeStage.DEAD or (w_a and not w_a.alive)) and a_id in living_ids:
                deaths += 1

    active_pop = sum(1 for a in self.agents.values() if a.stage != LifeStage.DEAD)
    final_world = WorldState.from_state(self.state)

    return PopulationSchedulerMetrics(
        ticks_elapsed=final_world.tick,
        total_transitions=transitions_count,
        births=births,
        deaths=deaths,
        teachings=teachings,
        archive_coverage=self.archive.coverage(),
        active_population=active_pop,
    )

    with a linear term to guarantee strict Pareto optimality (Hayes et al., AAMAS 2022):
        S(v, p) = -max_k { p_k * (z*_k - v_k) } + rho * sum_k (p_k * v_k) - cost

    Returns:
        Augmented Chebyshev utility scalar.

        p_k * (z_k - v_k)
        for p_k, z_k, v_k in zip(ratios, utopia, values, strict=True)
    )

    linear_term = sum(p_k * v_k for p_k, v_k in zip(ratios, values, strict=True))
    return -max_deficit + rho * linear_term - cost

    def decide(
    snapshot: BrainSnapshot,
    context: tuple[float, ...],
    candidates: tuple[Candidate, ...],
    tick: int,
    *,
    seed: int | None = None,
) -> BrainSnapshot:
    """Rank legal, age-accessible candidates using motive ratios, experience, and decision policy.

    Supports Tier-0 linear scalarization (default) and Tier-1 Augmented Chebyshev scalarization
    with UCB1 exploration and Mellowmax temperature sampling when configured via
    `snapshot.config.use_tier1_decision`.

    Args:
        snapshot: Current agent developmental brain snapshot.
        context: 8-D environmental context signal.
        candidates: Tuple of declared candidate actions.
        tick: Simulation clock tick.
        seed: Optional pseudo-random seed for stochastic exploration sampling. When None,
              deterministic highest-scoring candidate selection is used.

    Returns:
        Updated BrainSnapshot with pending Decision.
    """
    stage = stage_at(snapshot, tick)
    if stage == LifeStage.DEAD or snapshot.pending is not None:
        raise ValueError("Agent is dead or already awaiting an outcome")
    if not candidates or len(candidates) > snapshot.config.max_candidates:
        raise ValueError("Candidate budget must be nonempty and bounded")
    if len({item.action_id for item in candidates}) != len(candidates):
        raise ValueError("Candidate action IDs must be unique")
    age = tick - snapshot.birth.born_at
    activation, ratios = drive_step(
        snapshot.birth.genome, context, snapshot.activation, age / snapshot.config.lifespan,
    )
    eligible = tuple(item for item in candidates if item.permitted and item.minimum_age <= age)
    if not eligible:
        raise ValueError("No legal developmental action is available")

    config = snapshot.config
    use_tier1 = config.use_tier1_decision

    def evaluate_candidate(candidate: Candidate) -> float:
        estimate = next(
            (
                item
                for item in snapshot.knowledge
                if item.context_key == candidate.context_key
                and item.skill_key == candidate.skill_key
            ),
            None,
        )
        if estimate is None:
            values = candidate.predicted
            samples = 0
            lesson = next(
                (
                    item
                    for item in reversed(snapshot.lessons)
                    if item.estimate.context_key == candidate.context_key
                    and item.estimate.skill_key == candidate.skill_key
                ),
                None,
            )
            if lesson is not None:
                weight = config.advice_weight
                values = tuple(
                    (1.0 - weight) * value + weight * advice
                    for value, advice in zip(values, lesson.estimate.values, strict=True)
                )
        else:
            values = estimate.values
            samples = estimate.samples

        if use_tier1:
            ucb_values = compute_ucb_values(
                values,
                samples=samples,
                total_decisions=snapshot.sequence,
                exploration_c=config.exploration_c,
            )
            return chebyshev_score(
                ratios,
                ucb_values,
                candidate.cost,
                rho=config.chebyshev_rho,
            )

        return linear_score(ratios, values, candidate.cost)

    scores_by_id = {c.action_id: evaluate_candidate(c) for c in eligible}

    if seed is not None and use_tier1:
        candidate_scores = tuple(scores_by_id[c.action_id] for c in eligible)
        probs = mellowmax_probabilities(candidate_scores, beta=config.mellowmax_beta)
        rng = random.Random(seed)
        r = rng.random()
        cumulative = 0.0
        chosen = eligible[-1]
        for cand, p in zip(eligible, probs, strict=True):
            cumulative += p
            if r <= cumulative:
                chosen = cand
                break
    else:
        ranked = sorted(eligible, key=lambda item: (-scores_by_id[item.action_id], item.action_id))
        chosen = ranked[0]

    chosen_score = scores_by_id[chosen.action_id]
    decision = Decision(
        agent_id=snapshot.birth.agent_id,
        sequence=snapshot.sequence + 1,
        tick=tick,
        candidate=chosen,
        activation=activation,
        ratios=ratios,
        score=chosen_score,
    )
    return updated(
        snapshot,
        tick=tick,
        activation=activation,
        sequence=decision.sequence,
        pending=decision,
    )

    def mellowmax_probabilities(
    scores: tuple[float, ...],
    beta: float = 10.0,
) -> tuple[float, ...]:
    """Compute softmax exploration probabilities over candidate action scores.

    Args:
        scores: Tuple of candidate scalar utility scores.
        beta: Softmax temperature parameter (beta > 0).

    Returns:
        Probability distribution tuple summing to 1.0.
    """
    if not scores:
        return ()
    if beta <= 0.0:
        uniform_p = 1.0 / len(scores)
        return (uniform_p,) * len(scores)

    max_s = max(scores)
    exp_scores = [math.exp(min(500.0, max(-500.0, beta * (s - max_s)))) for s in scores]
    total_exp = sum(exp_scores)
    if total_exp <= 0.0:
        uniform_p = 1.0 / len(scores)
        return (uniform_p,) * len(scores)
    return tuple(p / total_exp for p in exp_scores)

    """Consume one trusted verified receipt matching the sole pending action.

    rewards here violates the caller contract even if the schema validates.

        raise ValueError("Learning needs a pending decision and verified outcome")

            or outcome.sequence != pending.sequence
            or outcome.action_id != pending.candidate.action_id):
        raise ValueError("Outcome does not match the pending decision")

        raise ValueError("Dead agents cannot learn")

        raise ValueError("Duplicate recent evidence")

        item for item in snapshot.knowledge

        (1.0 - rate) * old + rate * new

        context_key=candidate.context_key, skill_key=candidate.skill_key,
        values=values, samples=1 if prior is None else prior.samples + 1,

        knowledge=(remaining + (estimate,))[-snapshot.config.knowledge_limit:],

    class DevelopmentArchive:
    """Bounded archive, separate from the legacy five-drive archive.

    Descriptor dimensions and validation suite are fixed for the entire archive.
    A replacement never destroys the caller's original snapshot or run artifacts.
    """

    def __init__(
        self, suite_id: str, *, dimensions: int = 2, bins: int = 5,
        capacity: int = 25, min_gain: float = 0.0, max_retention_drop: float = 0.05,
    ) -> None:
        if not suite_id or len(suite_id) > 128:
            raise ValueError("A bounded validation-suite identifier is required")
        if any(isinstance(v, bool) or not isinstance(v, int) or v < 1
               for v in (dimensions, bins, capacity)):
            raise ValueError("Dimensions, bins and capacity must be positive integers")
        if dimensions > 8 or bins > 100 or capacity > 10000:
            raise ValueError("Archive budget exceeded")
        if any(not math.isfinite(v) or not 0 <= v <= 1 for v in (min_gain, max_retention_drop)):
            raise ValueError("Validation thresholds must lie in [0, 1]")
        self.suite_id = suite_id
        self.dimensions = dimensions
        self.bins = bins
        self.capacity = capacity
        self.min_gain = min_gain
        self.max_retention_drop = max_retention_drop
        self._cells: dict[tuple[int, ...], DevelopmentElite] = {}

    def consider(
        self, snapshot: BrainSnapshot, descriptor: tuple[float, ...],
        assessment: EvolutionAssessment,
    ) -> bool:
        """Accept externally evaluated improvements without exceeding capacity."""
        if len(descriptor) != self.dimensions or any(
            not math.isfinite(v) or not 0 <= v <= 1 for v in descriptor
        ):
            raise ValueError("Descriptor must match normalized archive dimensions")
        if assessment.agent_id != snapshot.birth.agent_id or assessment.suite_id != self.suite_id:
            raise ValueError("Assessment identity or validation suite mismatch")
        if snapshot.pending is not None or stage_at(snapshot, snapshot.tick) not in (
            LifeStage.ADULT, LifeStage.ELDER,
        ):
            raise ValueError("Archive candidates must be mature with settled actions")
        if (not assessment.verified or not assessment.safety_passed
                or assessment.retention_drop > self.max_retention_drop
                or assessment.score - assessment.baseline <= self.min_gain):
            return False
        cell = tuple(min(self.bins - 1, int(v * self.bins)) for v in descriptor)
        old = self._cells.get(cell)
        if old is None and len(self._cells) >= self.capacity:
            return False
        if old is not None and assessment.score <= old.assessment.score:
            return False
        self._cells[cell] = DevelopmentElite(snapshot, descriptor, assessment)
        return True

    def entries(self) -> tuple[DevelopmentElite, ...]:
        """Return elites in deterministic cell order for a host reproduction policy."""
        return tuple(self._cells[key] for key in sorted(self._cells))
