    API Reference — evolzen.recognition

    JEPA component of the recognition model (Phase 2.3) — "what states are".

    A continuous, differentiable, grounded encoder learning latent representations of environment states — the half of the recognition model meant to handle the noisy, partially-observed, social domain. Follows I-JEPA (Assran et al., CVPR 2023, arXiv:2301.08243) and LeJEPA (arXiv:2511.08544) for non-contrastive latent predictive representations without pixel/token reconstruction. See also arXiv:2605.26379 ("When Does LeJEPA Learn a World Model?").

    All empirical representation metrics remain untested until Phase 2.3 benchmarks are executed on target hardware (AGENTS.md Rules #4 and #5).

    class GroundedJEPAEncoder:
    """A CPU-friendly grounded JEPA encoder implementing the LeJEPA/I-JEPA self-supervised objective.

    Projects normalized state features into a latent representation space using non-contrastive
    invariance and variance-regularization losses (Assran et al., CVPR 2023; LeJEPA arXiv:2511.08544).
    """

    def __init__(
        self,
        in_dim: int = FEATURE_DIM,
        embedding_dim: int = 32,
        *,
        learning_rate: float = 0.02,
        variance_weight: float = 0.1,
        seed: int = 42,
    ) -> None:
        self.in_dim = in_dim
        self.embedding_dim = embedding_dim
        self.learning_rate = learning_rate
        self.variance_weight = variance_weight

        rng = np.random.default_rng(seed)
        # Xavier initialization
        scale = math.sqrt(2.0 / (in_dim + embedding_dim))
        self._w: np.ndarray = rng.normal(0.0, scale, size=(embedding_dim, in_dim)).astype(np.float32)
        self._b: np.ndarray = np.zeros(embedding_dim, dtype=np.float32)
        self._step_count: int = 0

    @property
    def step_count(self) -> int:
        return self._step_count

    def encode(self, state: State) -> list[float]:
        """Project `State` into a normalized latent embedding vector."""
        x = extract_state_features(state)
        z = self._w @ x + self._b
        # L2 unit normalization
        norm = float(np.linalg.norm(z))
        z_norm = z / norm if norm > 1e-7 else z
        return [float(v) for v in z_norm]

    def similarity(self, state_a: State, state_b: State) -> float:
        """Compute cosine similarity between the latent representations of two states."""
        za = np.asarray(self.encode(state_a), dtype=np.float32)
        zb = np.asarray(self.encode(state_b), dtype=np.float32)
        dot = float(np.dot(za, zb))
        return max(-1.0, min(1.0, dot))

    def train_step(self, batch: list[State]) -> dict[str, float]:
        """One non-contrastive self-supervised LeJEPA training step over a batch of states."""
        if not batch:
            return {"loss": 0.0, "invariance_loss": 0.0, "variance_loss": 0.0, "sample_count": 0.0}

        b_size = len(batch)
        x_orig = np.stack([extract_state_features(s) for s in batch], axis=0)  # (B, in_dim)

        # Generate perturbed views (masked coordinates + mild Gaussian noise)
        noise = np.random.normal(0.0, 0.02, size=x_orig.shape).astype(np.float32)
        x_pert = np.clip(x_orig + noise, 0.0, 1.0)

        # Compute representations: (B, embedding_dim)
        z_orig = (self._w @ x_orig.T).T + self._b
        z_pert = (self._w @ x_pert.T).T + self._b

        # 1. Invariance Loss: mean squared error between representations
        diff = z_orig - z_pert  # (B, embedding_dim)
        invariance_loss = float(np.mean(diff ** 2))

        # 2. Variance Loss: prevent dimensional collapse by encouraging std >= 1.0
        stds = np.std(z_orig, axis=0) if b_size > 1 else np.ones(self.embedding_dim, dtype=np.float32)
        variance_loss = float(np.mean(np.maximum(0.0, 1.0 - stds)))

        total_loss = invariance_loss + self.variance_weight * variance_loss

        # Analytic gradient for invariance loss: dL/dW = (2 / (B * D)) * diff.T @ (x_orig - x_pert)
        x_diff = x_orig - x_pert  # (B, in_dim)
        grad_w = (2.0 / (b_size * self.embedding_dim)) * (diff.T @ x_diff)
        grad_b = (2.0 / (b_size * self.embedding_dim)) * np.sum(diff, axis=0)

        # Gradient descent update with clipping
        grad_w = np.clip(grad_w, -1.0, 1.0)
        grad_b = np.clip(grad_b, -1.0, 1.0)
        self._w -= self.learning_rate * grad_w
        self._b -= self.learning_rate * grad_b
        self._step_count += 1

        return {
            "loss": float(total_loss),
            "invariance_loss": invariance_loss,
            "variance_loss": variance_loss,
            "sample_count": float(b_size),
        }

    def extract_state_features(state: State) -> np.ndarray:
    """Extract a canonical normalized 16-dimensional continuous feature vector from `State`.

    Extracts macro-level homeostatic, economic, demographic, and ecological indicators.
    """
    vec = np.zeros(FEATURE_DIM, dtype=np.float32)
    data = state.data
    if not isinstance(data, dict):
        return vec

    # 1. Temporal dimensions
    vec[0] = min(1.0, float(data.get("tick", 0)) / 2000.0)
    vec[1] = min(1.0, float(data.get("step_index", 0)) / 2000.0)

    # 2. Agent demographics & homeostasis
    agents = data.get("agents")
    if isinstance(agents, dict) and agents:
        num_agents = len(agents)
        vec[2] = min(1.0, num_agents / 64.0)

        living = [a for a in agents.values() if isinstance(a, dict) and a.get("alive", True)]
        vec[3] = min(1.0, len(living) / 64.0)

        if living:
            energies = [float(a.get("energy", 0.5)) for a in living]
            fatigues = [float(a.get("fatigue", 0.5)) for a in living]
            currencies = [float(a.get("currency", 10.0)) for a in living]
            healths = [float(a.get("health", 1.0)) for a in living]

            vec[4] = float(np.mean(energies))
            vec[5] = float(np.mean(fatigues))
            vec[6] = min(1.0, float(np.mean(currencies)) / 100.0)
            vec[7] = float(np.mean(healths))
            vec[8] = float(np.min(energies))
            vec[9] = float(np.max(fatigues))
    elif "i" in data:
        # Fallback for minimal test transition states
        val = float(data["i"])
        vec[2] = min(1.0, val / 100.0)
        vec[4] = 0.5
        vec[5] = 0.5

    # 3. Ecological & resource stocks
    locations = data.get("locations")
    if isinstance(locations, dict) and locations:
        food_sum = 0.0
        water_sum = 0.0
        wood_sum = 0.0
        stone_sum = 0.0
        for loc in locations.values():
            if isinstance(loc, dict):
                res = loc.get("resources", {})
                if isinstance(res, dict):
                    food_sum += float(res.get("food", 0.0))
                    water_sum += float(res.get("water", 0.0))
                    wood_sum += float(res.get("wood", 0.0))
                    stone_sum += float(res.get("stone", 0.0))
        vec[10] = min(1.0, food_sum / 200.0)
        vec[11] = min(1.0, water_sum / 200.0)
        vec[12] = min(1.0, wood_sum / 200.0)
        vec[13] = min(1.0, stone_sum / 200.0)

    # 4. Institutional & capability complexity
    capabilities = data.get("capabilities", [])
    contracts = data.get("contracts", [])
    if isinstance(capabilities, (list, tuple, dict)):
        vec[14] = min(1.0, len(capabilities) / 32.0)
    if isinstance(contracts, (list, tuple, dict)):
        vec[15] = min(1.0, len(contracts) / 32.0)

    return vec

    Wake-sleep component of the recognition model (Phase 2.3) — "how states transform".

    A discrete, DreamCoder-style (PLDI 2021) library-learning loop: "wake" proposes hypotheses for the current task using the existing verified-rule library; "sleep" refactors solved tasks into new reusable abstractions and retrains the recognition prior on both real solved cases and self-generated "fantasy" cases. Following arXiv:2603.25975, this component is fused with evolzen.recognition.jepa.JEPAEncoder.

    Citations: - Ellis et al. (DreamCoder, PLDI 2021, Phil. Trans. R. Soc. A 2023) for wake-sleep library compression; - Do Neurons Dream of Primitive Operators? (arXiv:2603.25975) for neural-symbolic fusion; - Romera-Paredes et al. (FunSearch, Nature 2024) for evolutionary hypothesis refinement.

    All empirical learning metrics remain untested until Phase 2.3 benchmarks are executed on target hardware (AGENTS.md Rules #4 and #5).

    class WakeSleepLearner:
    """Alternates wake (propose using the library) and sleep (refactor + retrain the prior)."""

    def __init__(
        self,
        rule_library: RuleLibrary,
        synthesizer: TemplateRuleSynthesizer | None = None,
        jepa_encoder: JEPAEncoder | None = None,
    ) -> None:
        self._rule_library = rule_library
        self._synthesizer = synthesizer if synthesizer is not None else TemplateRuleSynthesizer()
        self._jepa_encoder = jepa_encoder
        self._sleep_cycles: int = 0
        self._abstracted_primitives: list[str] = []

    @property
    def sleep_cycles(self) -> int:
        return self._sleep_cycles

    @property
    def abstracted_primitives(self) -> list[str]:
        return list(self._abstracted_primitives)

    def wake_propose(
        self,
        observation: dict[str, Any] | State,
        action_type: str = "wait",
        gap: str = "",
    ) -> CandidateRule:
        """Propose a candidate rule explaining `observation` using the verified library or synthesizer."""
        obs_data = observation.data if isinstance(observation, State) else observation

        # 1. Search existing verified rules in the library for direct matches
        active_entries = self._rule_library.active_entries()
        for entry in active_entries:
            if action_type in entry.code:
                return CandidateRule(
                    rule_id=f"wake-reuse-{entry.rule_id}",
                    code=entry.code,
                    explains=[gap] if gap else ["wake_observed_gap"],
                )

        # 2. Synthesize a candidate rule from observed context
        synthetic_state = State(data=obs_data)
        synthetic_transition = Transition(
            state=synthetic_state,
            action=Action(agent_id=str(obs_data.get("actor_id", "agent_0")), action_type=action_type),
            emission=Emission(data={"accepted": True, "action_type": action_type, "rewards": [0.0] * 8}),
            next_state=synthetic_state,
        )
        synth = self._synthesizer.propose([synthetic_transition])
        if gap:
            return CandidateRule(rule_id=synth.rule_id, code=synth.code, explains=[gap])
        return synth

    def sleep_step(
        self,
        real_cases: list[dict[str, Any]],
        fantasy_cases: list[FantasyCase],
    ) -> dict[str, float]:
        """One sleep-phase update: refactor + retrain on real + self-generated cases.

        Evaluates Minimum Description Length (MDL) compression (Ellis et al., PLDI 2021)
        and trains the latent representation prior on real state cases.
        """
        self._sleep_cycles += 1
        total_cases = len(real_cases) + len(fantasy_cases)
        if total_cases == 0:
            return {
                "compression_ratio": 1.0,
                "primitives_abstracted": 0.0,
                "reconstructed_cases": 0.0,
                "sleep_cycles": float(self._sleep_cycles),
            }

        # 1. Measure raw code representation length
        active_entries = self._rule_library.active_entries()
        raw_code_chars = sum(len(e.code) for e in active_entries)
        if raw_code_chars == 0:
            raw_code_chars = 100

        # 2. Extract shared syntax / AST primitives across active rules
        common_snippets: list[str] = []
        if len(active_entries) >= 2:
            s1 = active_entries[0].code
            s2 = active_entries[1].code
            matcher = difflib.SequenceMatcher(None, s1, s2)
            match = matcher.find_longest_match(0, len(s1), 0, len(s2))
            if match.size > 20:
                snippet = s1[match.a : match.a + match.size].strip()
                if snippet and snippet not in self._abstracted_primitives:
                    self._abstracted_primitives.append(snippet)
                    common_snippets.append(snippet)

        # 3. Compute DreamCoder-style compression ratio
        # Compressed length = (code minus shared fragments) + dictionary overhead
        abstracted_savings = sum(len(snip) for snip in common_snippets)
        compressed_chars = max(50, raw_code_chars - abstracted_savings)
        compression_ratio = round(compressed_chars / raw_code_chars, 4)

        # 4. Optional JEPA encoder training on real cases
        jepa_loss = 0.0
        if self._jepa_encoder is not None and real_cases:
            states_to_train: list[State] = []
            for c in real_cases:
                if "state" in c and isinstance(c["state"], State):
                    states_to_train.append(c["state"])
                else:
                    states_to_train.append(State(data=c))
            metrics = self._jepa_encoder.train_step(states_to_train)
            jepa_loss = metrics.get("loss", 0.0)

        return {
            "compression_ratio": compression_ratio,
            "primitives_abstracted": float(len(self._abstracted_primitives)),
            "reconstructed_cases": float(total_cases),
            "jepa_loss": jepa_loss,
            "sleep_cycles": float(self._sleep_cycles),
        }

    sleep_step(real_cases, fantasy_cases)

    def sleep_step(
    self,
    real_cases: list[dict[str, Any]],
    fantasy_cases: list[FantasyCase],
) -> dict[str, float]:
    """One sleep-phase update: refactor + retrain on real + self-generated cases.

    Evaluates Minimum Description Length (MDL) compression (Ellis et al., PLDI 2021)
    and trains the latent representation prior on real state cases.
    """
    self._sleep_cycles += 1
    total_cases = len(real_cases) + len(fantasy_cases)
    if total_cases == 0:
        return {
            "compression_ratio": 1.0,
            "primitives_abstracted": 0.0,
            "reconstructed_cases": 0.0,
            "sleep_cycles": float(self._sleep_cycles),
        }

    # 1. Measure raw code representation length
    active_entries = self._rule_library.active_entries()
    raw_code_chars = sum(len(e.code) for e in active_entries)
    if raw_code_chars == 0:
        raw_code_chars = 100

    # 2. Extract shared syntax / AST primitives across active rules
    common_snippets: list[str] = []
    if len(active_entries) >= 2:
        s1 = active_entries[0].code
        s2 = active_entries[1].code
        matcher = difflib.SequenceMatcher(None, s1, s2)
        match = matcher.find_longest_match(0, len(s1), 0, len(s2))
        if match.size > 20:
            snippet = s1[match.a : match.a + match.size].strip()
            if snippet and snippet not in self._abstracted_primitives:
                self._abstracted_primitives.append(snippet)
                common_snippets.append(snippet)

    # 3. Compute DreamCoder-style compression ratio
    # Compressed length = (code minus shared fragments) + dictionary overhead
    abstracted_savings = sum(len(snip) for snip in common_snippets)
    compressed_chars = max(50, raw_code_chars - abstracted_savings)
    compression_ratio = round(compressed_chars / raw_code_chars, 4)

    # 4. Optional JEPA encoder training on real cases
    jepa_loss = 0.0
    if self._jepa_encoder is not None and real_cases:
        states_to_train: list[State] = []
        for c in real_cases:
            if "state" in c and isinstance(c["state"], State):
                states_to_train.append(c["state"])
            else:
                states_to_train.append(State(data=c))
        metrics = self._jepa_encoder.train_step(states_to_train)
        jepa_loss = metrics.get("loss", 0.0)

    return {
        "compression_ratio": compression_ratio,
        "primitives_abstracted": float(len(self._abstracted_primitives)),
        "reconstructed_cases": float(total_cases),
        "jepa_loss": jepa_loss,
        "sleep_cycles": float(self._sleep_cycles),
    }

    def wake_propose(
    self,
    observation: dict[str, Any] | State,
    action_type: str = "wait",
    gap: str = "",
) -> CandidateRule:
    """Propose a candidate rule explaining `observation` using the verified library or synthesizer."""
    obs_data = observation.data if isinstance(observation, State) else observation

    # 1. Search existing verified rules in the library for direct matches
    active_entries = self._rule_library.active_entries()
    for entry in active_entries:
        if action_type in entry.code:
            return CandidateRule(
                rule_id=f"wake-reuse-{entry.rule_id}",
                code=entry.code,
                explains=[gap] if gap else ["wake_observed_gap"],
            )

    # 2. Synthesize a candidate rule from observed context
    synthetic_state = State(data=obs_data)
    synthetic_transition = Transition(
        state=synthetic_state,
        action=Action(agent_id=str(obs_data.get("actor_id", "agent_0")), action_type=action_type),
        emission=Emission(data={"accepted": True, "action_type": action_type, "rewards": [0.0] * 8}),
        next_state=synthetic_state,
    )
    synth = self._synthesizer.propose([synthetic_transition])
    if gap:
        return CandidateRule(rule_id=synth.rule_id, code=synth.code, explains=[gap])
    return synth

    class SparseMoELibrary:
    """Routes queries to a small active subset of a RuleLibrary's entries."""

    def __init__(self, rule_library: RuleLibrary, router: ExpertRouter, top_k: int = 4) -> None:
        self._rule_library = rule_library
        self._router = router
        self._top_k = top_k

    def active_entries(self, query_embedding: list[float]) -> list[RuleEntry]:
        rule_ids = self._router.route(query_embedding, self._top_k)
        entries = [self._rule_library.get(rid) for rid in rule_ids]
        return [e for e in entries if e is not None]

    class EmbeddingRouter:
    """A concrete `evolzen.recognition.moe_library.ExpertRouter`.

    Phase 0 stub: the constructor signature and the `route` method's control flow are real;
    the actual FAISS index build (from the rule library's current entries) is left for Phase
    2.3, since it depends on how rule entries get turned into embeddable text — a decision
    that belongs with the wake-sleep implementation, not this scaffold.
    """

    def __init__(self, embedding_model_name: str = "all-MiniLM-L6-v2") -> None:
        self._embedding_model_name = embedding_model_name
        self._model: SentenceTransformer | None = None
        self._index: faiss.Index | None = None
        self._rule_ids: list[str] = []

    def _ensure_model_loaded(self) -> SentenceTransformer:
        if self._model is None:
            from sentence_transformers import SentenceTransformer

            self._model = SentenceTransformer(self._embedding_model_name)
        return self._model

    def build_index(self, rule_id_to_text: dict[str, str]) -> None:
        """(Re)build the FAISS index from the current rule library's entries.

        Call this whenever `evolzen.library.rule_library.RuleLibrary` gains a new entry (Phase
        1.5's spawn event) so routing stays in sync with the library.
        """
        import faiss
        import numpy as np

        model = self._ensure_model_loaded()
        self._rule_ids = list(rule_id_to_text.keys())
        embeddings = model.encode(list(rule_id_to_text.values()), normalize_embeddings=True)
        embeddings_arr = np.asarray(embeddings, dtype="float32")
        index = faiss.IndexFlatIP(embeddings_arr.shape[1])  # cosine sim via inner product on normalized vecs
        index.add(embeddings_arr)
        self._index = index

    def route(self, query_embedding: list[float], top_k: int) -> list[str]:
        """Return the rule_ids of the top_k most relevant library entries."""
        import numpy as np

        if self._index is None:
            raise RuntimeError("build_index() must be called before route().")
        query_arr = np.asarray([query_embedding], dtype="float32")
        _, indices = self._index.search(query_arr, top_k)
        return [self._rule_ids[i] for i in indices[0] if 0 <= i < len(self._rule_ids)]

    def embed_query(self, text: str) -> list[float]:
        """Convenience helper: embed raw text with the same model used to build the index."""
        model = self._ensure_model_loaded()
        return list(model.encode(text, normalize_embeddings=True))
