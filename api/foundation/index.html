    API Reference — shared foundation

    Repo-wide configuration.

    One EvolZenSettings root, composed of one nested settings group per subsystem that actually needs configuration, loaded from:

    1. CLI flags (highest precedence — set by evolzen.cli)
    2. Environment variables, prefixed EVOLZEN_ (e.g. EVOLZEN_AGENTS__POPULATION_SIZE=15, using pydantic-settings' nested-delimiter convention)
    3. The YAML files under configs/ (e.g. configs/agents.yaml)
    4. The class defaults below

    Layers 2-4 are handled by load_settings below; layer 1 is evolzen.cli's job.

    def load_settings(settings_cls: type[_SettingsT] = EvolZenSettings, yaml_path: Path | None = None) -> _SettingsT:  # type: ignore[assignment]
    """Construct settings, layering a YAML file under environment-variable overrides.

    Args:
        settings_cls: defaults to `EvolZenSettings`; a specific subsystem group can be passed
            for isolated construction in tests.
        yaml_path: path to a YAML file (e.g. ``configs/agents.yaml``). If ``None`` or the file
            doesn't exist, this is equivalent to ``settings_cls()``.
    """
    yaml_overrides: dict[str, Any] = {}
    if yaml_path is not None and yaml_path.exists():
        loaded = yaml.safe_load(yaml_path.read_text())
        if loaded:
            yaml_overrides = loaded
    return settings_cls(**yaml_overrides)

    def configure_logging(*, level: str = "INFO", phase: str | None = None) -> None:
    """Configure the shared loguru logger.

    Args:
        level: standard logging level name.
        phase: optional phase tag (e.g. ``"1.4"``) bound into every log line from this
            process, so logs from a multi-phase run can be filtered by phase.
    """
    global _CONFIGURED
    if _CONFIGURED:
        return
    logger.remove()
    fmt = "<green>{time:HH:mm:ss}</green> | <level>{level: <8}</level> | "
    if phase:
        fmt += f"<cyan>phase={phase}</cyan> | "
    fmt += "<level>{message}</level>"
    logger.add(sys.stderr, level=level, format=fmt, colorize=True)
    _CONFIGURED = True

    class WandbTracker:
    """Weights & Biases-backed tracker. Requires the `evolzen[tracking]` optional extra.

    Import of `wandb` is deferred to `__init__` so importing this module never requires the
    extra — only *constructing* a `WandbTracker` does.
    """

    def __init__(self, project: str, run_name: str | None = None) -> None:
        try:
            import wandb
        except ImportError as exc:  # pragma: no cover - exercised only without the extra
            raise ImportError(
                "WandbTracker requires the 'tracking' extra: uv sync --extra tracking"
            ) from exc
        self._run: Any = wandb.init(project=project, name=run_name)

    def log(self, metrics: dict[str, float], *, step: int | None = None) -> None:
        self._run.log(metrics, step=step)

    def log_params(self, params: dict[str, Any]) -> None:
        self._run.config.update(params)

    def finish(self) -> None:
        self._run.finish()

    The interactive transition-system schema (Phase 1.1).

    evolzen.environment is formalized as a seeded, typed interactive transition system — states, typed actions, explicit exogenous inputs, seeds, and typed emissions — following Tycho (arXiv:2607.28287) and ADR 0005. Every outcome logged through this schema must be independently re-derivable from (state, action, exogenous_input, seed) alone. These contracts are an EvolZen design choice and remain empirically untested until Phase 1.1 experiments are run.

    Phase 1.2 — Unified Agent Profile and Identity Schema.

    Defines the persistent social, institutional, and lineage identity contracts for agents in the EvolZen multi-agent population (docs/ARCHITECTURE.md §3.2, docs/phases/1.2-agents.md, ADR 0006, ADR 0008). Replaces the legacy five-coordinate DriveVector with a rich social profile linked to evolzen.schemas.world.WorldAgentState and the eight-coordinate developmental core (evolzen.schemas.development.AgentDevelopmentState).

    Citations: - Park et al. (Generative Agents, arXiv:2304.03442) for agent social identity, memory, and roles; - Cook et al. (Artificial Generational Intelligence, arXiv:2406.00392) for cultural transfer; - Vassiliades et al. (CVT-MAP-Elites, IEEE TEVC 2018, DOI:10.1109/TEVC.2017.2735550) for behavioral descriptor keys in high-dimensional phenotype archives.

    These contracts are an EvolZen specification choice and remain empirically untested until Phase 1.2 population experiments are run.

    """Validate restoration boundaries; authenticity still belongs to the host."""
    if self.tick < self.birth.born_at:
        raise ValueError("Clock predates birth")
    if len(self.knowledge) > self.config.knowledge_limit:
        raise ValueError("Knowledge budget exceeded")

        raise ValueError("Memory budget exceeded")
    keys = [(item.context_key, item.skill_key) for item in self.knowledge]
    if len(set(keys)) != len(keys):
        raise ValueError("Duplicate knowledge keys")
    if self.pending is not None and (
        self.pending.agent_id != self.birth.agent_id
        or self.pending.sequence != self.sequence
        or self.pending.tick != self.tick
        or self.pending.activation != self.activation
    ):
        raise ValueError("Pending decision does not match snapshot")
    for item in self.recent:
        if (item.agent_id != self.birth.agent_id or not item.verified
                or item.tick > self.tick or item.sequence > self.sequence):
            raise ValueError("Invalid outcome history")
    for lesson in self.lessons:
        if (lesson.learner_id != self.birth.agent_id
                or lesson.teacher_id not in self.birth.parents
                or not self.birth.born_at <= lesson.taught_at <= self.tick):
            raise ValueError("Invalid lesson history")
    return self

    class TransitionLogStore:
    """Append-only DuckDB store for transitions, snapshots, growth events, and branches.

    Every record is serialized strictly as JSON (never pickle). Appends run inside
    explicit DuckDB transactions and extend a per-store SHA-256 hash chain.
    """

    def __init__(self, db_path: Path) -> None:
        self._db_path = db_path
        self._db_path.parent.mkdir(parents=True, exist_ok=True)
        self._conn: duckdb.DuckDBPyConnection = duckdb.connect(str(self._db_path))
        self._init_schema()

    def _init_schema(self) -> None:
        self._conn.execute("BEGIN TRANSACTION")
        try:
            self._conn.execute(
                """
                CREATE TABLE IF NOT EXISTS schema_meta (
                    schema_version INTEGER PRIMARY KEY,
                    created_at TIMESTAMP NOT NULL
                )
                """
            )
            self._conn.execute(
                """
                CREATE TABLE IF NOT EXISTS transitions (
                    seq_no BIGINT NOT NULL,
                    transition_id VARCHAR PRIMARY KEY,
                    run_id VARCHAR NOT NULL,
                    branch_id VARCHAR NOT NULL,
                    tick BIGINT NOT NULL,
                    timestamp TIMESTAMP NOT NULL,
                    seed BIGINT NOT NULL,
                    state_json VARCHAR NOT NULL,
                    action_json VARCHAR NOT NULL,
                    exogenous_json VARCHAR NOT NULL,
                    emission_json VARCHAR NOT NULL,
                    next_state_json VARCHAR NOT NULL,
                    prev_hash VARCHAR NOT NULL,
                    entry_hash VARCHAR NOT NULL
                )
                """
            )
            self._conn.execute(
                """
                CREATE TABLE IF NOT EXISTS snapshots (
                    snapshot_id VARCHAR PRIMARY KEY,
                    run_id VARCHAR NOT NULL,
                    branch_id VARCHAR NOT NULL,
                    tick BIGINT NOT NULL,
                    parent_snapshot_id VARCHAR,
                    world_version VARCHAR NOT NULL,
                    snapshot_json VARCHAR NOT NULL,
                    snapshot_hash VARCHAR NOT NULL,
                    created_at TIMESTAMP NOT NULL
                )
                """
            )
            self._conn.execute(
                """
                CREATE TABLE IF NOT EXISTS growth_events (
                    seq_no BIGINT NOT NULL,
                    event_id VARCHAR PRIMARY KEY,
                    run_id VARCHAR NOT NULL,
                    branch_id VARCHAR NOT NULL,
                    tick BIGINT NOT NULL,
                    category VARCHAR NOT NULL,
                    attribution VARCHAR NOT NULL,
                    actor_id VARCHAR NOT NULL,
                    entity_id VARCHAR NOT NULL,
                    summary VARCHAR NOT NULL,
                    transition_id VARCHAR NOT NULL
                )
                """
            )
            self._conn.execute(
                """
                CREATE TABLE IF NOT EXISTS branches (
                    branch_id VARCHAR NOT NULL,
                    run_id VARCHAR NOT NULL,
                    parent_branch_id VARCHAR NOT NULL,
                    fork_tick BIGINT NOT NULL,
                    fork_snapshot_id VARCHAR NOT NULL,
                    created_at TIMESTAMP NOT NULL,
                    PRIMARY KEY (run_id, branch_id)
                )
                """
            )
            existing_version = self._conn.execute(
                "SELECT schema_version FROM schema_meta WHERE schema_version = ?",
                [SCHEMA_VERSION],
            ).fetchone()
            if existing_version is None:
                self._conn.execute(
                    "INSERT INTO schema_meta (schema_version, created_at) VALUES (?, ?)",
                    [SCHEMA_VERSION, datetime.now(UTC).replace(tzinfo=None)],
                )
            self._conn.execute("COMMIT")
        except Exception:
            self._conn.execute("ROLLBACK")
            raise

    def close(self) -> None:
        """Close the underlying DuckDB connection."""
        self._conn.close()

    def __enter__(self) -> TransitionLogStore:
        return self

    def __exit__(
        self,
        exc_type: type[BaseException] | None,
        exc_val: BaseException | None,
        exc_tb: TracebackType | None,
    ) -> None:
        self.close()

    def schema_version(self) -> int:
        """Return the active persisted schema version."""
        row = self._conn.execute("SELECT MAX(schema_version) FROM schema_meta").fetchone()
        return int(row[0]) if row and row[0] is not None else 0

    def append(
        self,
        transition: Transition,
        *,
        run_id: str | None = None,
        branch_id: str | None = None,
    ) -> None:
        """Persist one transition atomically with hash-chain linkage."""
        resolved_run = run_id or str(transition.state.data.get("run_id", "default-run"))
        resolved_branch = branch_id or str(transition.state.data.get("branch_id", "main"))
        tick = int(transition.state.data.get("tick", transition.exogenous_input.tick))
        state_json = transition.state.model_dump_json()
        action_json = transition.action.model_dump_json()
        exogenous_json = transition.exogenous_input.model_dump_json()
        emission_json = transition.emission.model_dump_json()
        next_state_json = transition.next_state.model_dump_json()
        transition_id = str(transition.transition_id)
        ts = transition.timestamp.astimezone(UTC).replace(tzinfo=None)

        self._conn.execute("BEGIN TRANSACTION")
        try:
            last = self._conn.execute(
                "SELECT seq_no, entry_hash FROM transitions ORDER BY seq_no DESC LIMIT 1"
            ).fetchone()
            seq_no = int(last[0]) + 1 if last is not None else 1
            prev_hash = str(last[1]) if last is not None else _GENESIS_HASH
            entry_hash = _transition_digest(
                prev_hash,
                transition_id,
                resolved_run,
                resolved_branch,
                tick,
                transition.seed,
                state_json,
                action_json,
                exogenous_json,
                emission_json,
                next_state_json,
            )
            self._conn.execute(
                """
                INSERT INTO transitions (
                    seq_no, transition_id, run_id, branch_id, tick, timestamp, seed,
                    state_json, action_json, exogenous_json, emission_json, next_state_json,
                    prev_hash, entry_hash
                ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                """,
                [
                    seq_no,
                    transition_id,
                    resolved_run,
                    resolved_branch,
                    tick,
                    ts,
                    transition.seed,
                    state_json,
                    action_json,
                    exogenous_json,
                    emission_json,
                    next_state_json,
                    prev_hash,
                    entry_hash,
                ],
            )
            self._conn.execute("COMMIT")
        except Exception:
            self._conn.execute("ROLLBACK")
            raise

    def count(self, *, run_id: str | None = None, branch_id: str | None = None) -> int:
        """Number of transitions currently stored, optionally filtered by run/branch."""
        clauses: list[str] = []
        params: list[Any] = []
        if run_id is not None:
            clauses.append("run_id = ?")
            params.append(run_id)
        if branch_id is not None:
            clauses.append("branch_id = ?")
            params.append(branch_id)
        where = f" WHERE {' AND '.join(clauses)}" if clauses else ""
        row = self._conn.execute(f"SELECT COUNT(*) FROM transitions{where}", params).fetchone()
        return int(row[0]) if row is not None else 0

    def list_transitions(
        self,
        *,
        run_id: str | None = None,
        branch_id: str | None = None,
        from_tick: int | None = None,
        to_tick: int | None = None,
    ) -> list[Transition]:
        """Load transitions in append order, optionally filtered by run/branch/ticks."""
        clauses: list[str] = []
        params: list[Any] = []
        if run_id is not None:
            clauses.append("run_id = ?")
            params.append(run_id)
        if branch_id is not None:
            clauses.append("branch_id = ?")
            params.append(branch_id)
        if from_tick is not None:
            clauses.append("tick >= ?")
            params.append(from_tick)
        if to_tick is not None:
            clauses.append("tick <= ?")
            params.append(to_tick)
        where = f" WHERE {' AND '.join(clauses)}" if clauses else ""
        rows = self._conn.execute(
            f"""
            SELECT transition_id, timestamp, seed, state_json, action_json,
                   exogenous_json, emission_json, next_state_json
            FROM transitions{where}
            ORDER BY seq_no ASC
            """,
            params,
        ).fetchall()
        results: list[Transition] = []
        for row in rows:
            ts_val = row[1]
            if isinstance(ts_val, datetime) and ts_val.tzinfo is None:
                ts_val = ts_val.replace(tzinfo=UTC)
            results.append(
                Transition(
                    transition_id=row[0],
                    timestamp=ts_val,
                    seed=int(row[2]),
                    state=State.model_validate_json(row[3]),
                    action=Action.model_validate_json(row[4]),
                    exogenous_input=ExogenousInput.model_validate_json(row[5]),
                    emission=Emission.model_validate_json(row[6]),
                    next_state=State.model_validate_json(row[7]),
                )
            )
        return results

    def verify_integrity(self) -> bool:
        """Verify the SHA-256 hash chain across all persisted transitions."""
        rows = self._conn.execute(
            """
            SELECT seq_no, transition_id, run_id, branch_id, tick, seed,
                   state_json, action_json, exogenous_json, emission_json, next_state_json,
                   prev_hash, entry_hash
            FROM transitions
            ORDER BY seq_no ASC
            """
        ).fetchall()
        expected_prev = _GENESIS_HASH
        for row in rows:
            if row[11] != expected_prev:
                return False
            recomputed = _transition_digest(
                str(row[11]),
                str(row[1]),
                str(row[2]),
                str(row[3]),
                int(row[4]),
                int(row[5]),
                str(row[6]),
                str(row[7]),
                str(row[8]),
                str(row[9]),
                str(row[10]),
            )
            if recomputed != row[12]:
                return False
            expected_prev = str(row[12])
        return True

    def _insert_snapshot(self, snapshot: WorldSnapshot) -> None:
        payload = snapshot.model_dump_json()
        self._conn.execute(
            """
            INSERT INTO snapshots (
                snapshot_id, run_id, branch_id, tick, parent_snapshot_id,
                world_version, snapshot_json, snapshot_hash, created_at
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            """,
            [
                snapshot.snapshot_id,
                snapshot.run_id,
                snapshot.branch_id,
                snapshot.tick,
                snapshot.parent_snapshot_id,
                snapshot.world_version,
                payload,
                snapshot.state_hash,
                datetime.now(UTC).replace(tzinfo=None),
            ],
        )

    def _insert_branch(self, branch: BranchRecord) -> None:
        self._conn.execute(
            """
            INSERT INTO branches (
                branch_id, run_id, parent_branch_id, fork_tick, fork_snapshot_id, created_at
            ) VALUES (?, ?, ?, ?, ?, ?)
            """,
            [
                branch.branch_id,
                branch.run_id,
                branch.parent_branch_id,
                branch.fork_tick,
                branch.fork_snapshot_id,
                datetime.now(UTC).replace(tzinfo=None),
            ],
        )

    def save_snapshot(self, snapshot: WorldSnapshot) -> None:
        """Persist an immutable `WorldSnapshot` record atomically."""
        self._conn.execute("BEGIN TRANSACTION")
        try:
            self._insert_snapshot(snapshot)
            self._conn.execute("COMMIT")
        except Exception:
            self._conn.execute("ROLLBACK")
            raise

    def save_snapshot_and_branch(self, snapshot: WorldSnapshot, branch: BranchRecord) -> None:
        """Persist a fork `WorldSnapshot` and its `BranchRecord` in one atomic transaction."""
        self._conn.execute("BEGIN TRANSACTION")
        try:
            self._insert_snapshot(snapshot)
            self._insert_branch(branch)
            self._conn.execute("COMMIT")
        except Exception:
            self._conn.execute("ROLLBACK")
            raise

    def load_snapshot(self, snapshot_id: str) -> WorldSnapshot:
        """Load and validate a persisted `WorldSnapshot` by ID."""
        row = self._conn.execute(
            "SELECT snapshot_json, snapshot_hash FROM snapshots WHERE snapshot_id = ?",
            [snapshot_id],
        ).fetchone()
        if row is None:
            raise KeyError(f"Unknown snapshot_id: {snapshot_id}")
        snapshot = WorldSnapshot.model_validate_json(row[0])
        if snapshot.state.state_hash() != row[1]:
            raise ValueError(f"Snapshot hash mismatch for {snapshot_id}")
        return snapshot

    def latest_snapshot(
        self,
        run_id: str,
        branch_id: str = "main",
        *,
        max_tick: int | None = None,
    ) -> WorldSnapshot | None:
        """Return the latest snapshot for `(run_id, branch_id)` at or below `max_tick`."""
        clauses = ["run_id = ?", "branch_id = ?"]
        params: list[Any] = [run_id, branch_id]
        if max_tick is not None:
            clauses.append("tick <= ?")
            params.append(max_tick)
        row = self._conn.execute(
            f"""
            SELECT snapshot_json, snapshot_hash
            FROM snapshots
            WHERE {' AND '.join(clauses)}
            ORDER BY tick DESC, rowid DESC
            LIMIT 1
            """,
            params,
        ).fetchone()
        if row is None:
            return None
        snapshot = WorldSnapshot.model_validate_json(row[0])
        if snapshot.state.state_hash() != row[1]:
            raise ValueError(f"Snapshot hash mismatch for {snapshot.snapshot_id}")
        return snapshot

    def append_growth_events(
        self,
        events: list[GrowthEvent] | tuple[GrowthEvent, ...],
        *,
        run_id: str,
        branch_id: str = "main",
    ) -> None:
        """Persist attributable world-growth events atomically."""
        if not events:
            return
        self._conn.execute("BEGIN TRANSACTION")
        try:
            row = self._conn.execute("SELECT COALESCE(MAX(seq_no), 0) FROM growth_events").fetchone()
            seq = int(row[0]) if row is not None else 0
            for event in events:
                seq += 1
                self._conn.execute(
                    """
                    INSERT INTO growth_events (
                        seq_no, event_id, run_id, branch_id, tick, category,
                        attribution, actor_id, entity_id, summary, transition_id
                    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                    """,
                    [
                        seq,
                        event.event_id,
                        run_id,
                        branch_id,
                        event.tick,
                        event.category.value,
                        event.attribution.value,
                        event.actor_id,
                        event.entity_id,
                        event.summary,
                        event.transition_id,
                    ],
                )
            self._conn.execute("COMMIT")
        except Exception:
            self._conn.execute("ROLLBACK")
            raise

    def append_growth_events(
    self,
    events: list[GrowthEvent] | tuple[GrowthEvent, ...],
    *,
    run_id: str,
    branch_id: str = "main",
) -> None:
    """Persist attributable world-growth events atomically."""
    if not events:
        return
    self._conn.execute("BEGIN TRANSACTION")
    try:
        row = self._conn.execute("SELECT COALESCE(MAX(seq_no), 0) FROM growth_events").fetchone()
        seq = int(row[0]) if row is not None else 0
        for event in events:
            seq += 1
            self._conn.execute(
                """
                INSERT INTO growth_events (
                    seq_no, event_id, run_id, branch_id, tick, category,
                    attribution, actor_id, entity_id, summary, transition_id
                ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                """,
                [
                    seq,
                    event.event_id,
                    run_id,
                    branch_id,
                    event.tick,
                    event.category.value,
                    event.attribution.value,
                    event.actor_id,
                    event.entity_id,
                    event.summary,
                    event.transition_id,
                ],
            )
        self._conn.execute("COMMIT")
    except Exception:
        self._conn.execute("ROLLBACK")
        raise
