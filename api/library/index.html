    API Reference — evolzen.library

    The append-only, non-overwriting verified-rule library (Phase 1.5).

    Verified rules (that passed evolzen.exploration.verifier.Verifier) become frozen :class:~evolzen.schemas.library.RuleEntry instances stored here. A new entry is spawned only when no existing active rule is sufficiently similar to the candidate — the same "does this fit anything I already have?" logic as a Dirichlet process mixture model (Ferguson, 1973). Existing entries are never overwritten; the Phase 1.5 success metric is exactly zero forgetting on a held-out regression bank across repeated additions (untested until CI runs).

    Default routing uses difflib.SequenceMatcher on code text — a stdlib-only, dependency-free baseline. Callers may supply a custom routing_confidence_fn for embedding-based or AST-based similarity.

    Phase 1.5 — evolzen.library. Reference: docs/phases/1.5-library.md, docs/ARCHITECTURE.md §3.5, PRD FR-6, Ferguson (1973) "A Bayesian Analysis of Some Nonparametric Problems".

    class RuleLibrary:
    """Append-only collection of verified rules backed by :class:`~evolzen.library.persistence.LibraryStore`.

    All mutation methods write through the store and emit an :class:`~evolzen.schemas.library.AuditEvent`.
    No entry is ever overwritten or deleted.

    Args:
        store: The :class:`~evolzen.library.persistence.LibraryStore` instance.
        audit_log: The :class:`~evolzen.library.audit.AuditLog` instance.
        routing_threshold: Similarity threshold in ``[0.0, 1.0]`` above which a
            candidate is routed to an existing entry rather than spawning a new one.
        routing_confidence_fn: Similarity scorer.  Defaults to
            :func:`_default_confidence` (SequenceMatcher on code text).

    Example::

        store = LibraryStore(Path("data/library.duckdb"))
        audit = AuditLog(Path("data/library_audit.jsonl"))
        library = RuleLibrary(store, audit, routing_threshold=0.85)
        entry = library.spawn_if_novel(candidate, validity=ctx, evidence=ev)
    """

    def __init__(
        self,
        store: LibraryStore,
        audit_log: AuditLog,
        routing_threshold: float = 0.85,
        routing_confidence_fn: RoutingConfidenceFn | None = None,
    ) -> None:
        self._store = store
        self._audit = audit_log
        self._threshold = routing_threshold
        self._confidence_fn: RoutingConfidenceFn = (
            routing_confidence_fn or _default_confidence
        )

    # ------------------------------------------------------------------
    # Spawn / routing
    # ------------------------------------------------------------------

    def spawn_if_novel(
        self,
        candidate: CandidateRule,
        validity: ValidityContext,
        evidence: VerificationEvidence,
        provenance: ProvenanceClass = ProvenanceClass.EXPLORATION_VERIFIED,
    ) -> RuleEntry | None:
        """Spawn a new library entry if *candidate* is sufficiently novel.

        Computes similarity between *candidate* (wrapped as a temporary
        :class:`RuleEntry`) and each active entry.  If the maximum similarity
        is at or above ``routing_threshold``, the candidate is *routed* to the
        existing entry and ``None`` is returned.  Otherwise a new entry is
        appended and returned.

        No existing entry is ever modified.

        Args:
            candidate: The synthesiser hypothesis to evaluate.
            validity: Validity context under which the candidate was verified.
            evidence: Verification evidence for the candidate.
            provenance: Origin classification (default:
                :attr:`~evolzen.schemas.library.ProvenanceClass.EXPLORATION_VERIFIED`).

        Returns:
            The new :class:`RuleEntry` if spawned, or ``None`` if routed.
        """
        # Build a temporary entry just for similarity scoring (not persisted).
        temp_entry = _candidate_to_entry(candidate, provenance, validity, evidence)

        active = self._store.get_active()
        best_score = 0.0
        best_match: RuleEntry | None = None
        for existing in active:
            score = self._confidence_fn(temp_entry, existing)
            if score > best_score:
                best_score = score
                best_match = existing

        if best_score >= self._threshold and best_match is not None:
            # Route — record audit event, do NOT persist temp_entry.
            self._audit.record(
                make_event(
                    AuditEventType.ROUTE,
                    rule_id=best_match.rule_id,
                    detail={
                        "candidate_rule_id": candidate.rule_id,
                        "similarity_score": best_score,
                        "threshold": self._threshold,
                    },
                )
            )
            return None

        # Spawn — persist the new entry and record audit event.
        self._store.append_rule(temp_entry)
        self._audit.record(
            make_event(
                AuditEventType.SPAWN,
                rule_id=temp_entry.rule_id,
                detail={
                    "candidate_rule_id": candidate.rule_id,
                    "best_similarity": best_score,
                    "provenance": str(provenance),
                },
            )
        )
        return temp_entry

    # ------------------------------------------------------------------
    # Lifecycle transitions
    # ------------------------------------------------------------------

    def promote(self, rule_id: UUID, evidence: VerificationEvidence) -> RuleEntry:
        """Promote a rule by updating its verification evidence (stays ACTIVE).

        Records a PROMOTE audit event.  The rule must already be ACTIVE.

        Args:
            rule_id: UUID of the rule to promote.
            evidence: Updated :class:`~evolzen.schemas.library.VerificationEvidence`.

        Returns:
            The updated :class:`RuleEntry` (freshly read from the store).

        Raises:
            KeyError: If *rule_id* does not exist in the store.
            ValueError: If the rule is not currently ACTIVE.
        """
        entry = self._store.get_by_id(rule_id)
        if entry is None:
            raise KeyError(f"No rule entry found for rule_id={rule_id}")
        if entry.status is not RuleStatus.ACTIVE:
            raise ValueError(
                f"Only ACTIVE rules can be promoted; {rule_id} is {entry.status}"
            )
        updated = self._store.update_evidence(rule_id, evidence)
        self._audit.record(
            make_event(
                AuditEventType.PROMOTE,
                rule_id=rule_id,
                detail={"evidence_id": str(evidence.evidence_id)},
            )
        )
        return updated

    def supersede(
        self,
        old_id: UUID,
        new_entry: RuleEntry,
        link: SupersessionLink,
    ) -> None:
        """Supersede *old_id* with *new_entry*, linked by *link*.

        Persists *new_entry*, updates *old_id*'s status to SUPERSEDED, and
        appends the :class:`~evolzen.schemas.library.SupersessionLink`.
        Records a SUPERSEDE audit event.

        Args:
            old_id: UUID of the rule being superseded.
            new_entry: Replacement :class:`RuleEntry` (already constructed by caller).
            link: :class:`~evolzen.schemas.library.SupersessionLink` documenting the
                reason.

        Raises:
            KeyError: If *old_id* does not exist.
            ValueError: If ``link.old_rule_id != old_id`` or
                ``link.new_rule_id != new_entry.rule_id``.
        """
        if link.old_rule_id != old_id:
            raise ValueError("link.old_rule_id must equal old_id")
        if link.new_rule_id != new_entry.rule_id:
            raise ValueError("link.new_rule_id must equal new_entry.rule_id")
        old = self._store.get_by_id(old_id)
        if old is None:
            raise KeyError(f"No rule entry found for rule_id={old_id}")

        self._store.append_rule(new_entry)
        self._store.update_status(
            old_id,
            RuleStatus.SUPERSEDED,
            superseded_by=new_entry.rule_id,
            reason=None,
        )
        self._store.append_supersession(link)
        self._audit.record(
            make_event(
                AuditEventType.SUPERSEDE,
                rule_id=old_id,
                detail={
                    "new_rule_id": str(new_entry.rule_id),
                    "reason": link.reason,
                },
            )
        )

    def deprecate(self, rule_id: UUID, reason: str) -> RuleEntry:
        """Mark a rule as DEPRECATED with a documented reason.

        Records a DEPRECATE audit event.

        Args:
            rule_id: UUID of the rule to deprecate.
            reason: Human-readable deprecation reason.

        Returns:
            The updated :class:`RuleEntry`.

        Raises:
            KeyError: If *rule_id* does not exist.
        """
        entry = self._store.get_by_id(rule_id)
        if entry is None:
            raise KeyError(f"No rule entry found for rule_id={rule_id}")
        self._store.update_status(
            rule_id, RuleStatus.DEPRECATED, superseded_by=None, reason=reason
        )
        self._audit.record(
            make_event(
                AuditEventType.DEPRECATE,
                rule_id=rule_id,
                detail={"reason": reason},
            )
        )
        result = self._store.get_by_id(rule_id)
        assert result is not None
        return result

    def reject(
        self,
        candidate: CandidateRule,
        reason: str,
        evidence: VerificationEvidence | None = None,
    ) -> AuditEvent:
        """Record an explicit rejection decision for a candidate rule.

        Does not persist *candidate* into ``rule_entries``; appends an
        ``AuditEventType.REJECT`` event to the audit log.

        Args:
            candidate: Rejected synthesiser candidate.
            reason: Non-empty justification for rejection.
            evidence: Optional verification evidence associated with the check.

        Returns:
            The recorded :class:`~evolzen.schemas.library.AuditEvent`.

        Raises:
            ValueError: If *reason* is empty/whitespace.
        """
        if not reason.strip():
            raise ValueError("Rejection reason must be non-empty")
        try:
            parsed_id = UUID(candidate.rule_id)
        except ValueError:
            parsed_id = uuid4()
        detail: dict[str, object] = {
            "candidate_rule_id": candidate.rule_id,
            "reason": reason,
            "explains": list(candidate.explains),
        }
        if evidence is not None:
            detail["evidence_id"] = str(evidence.evidence_id)
            detail["false_promotion_rate"] = evidence.false_promotion_rate
        event = make_event(AuditEventType.REJECT, rule_id=parsed_id, detail=detail)
        self._audit.record(event)
        return event

    def revert_supersession(self, old_id: UUID, reason: str) -> RuleEntry:
        """Explicitly reverse a supersession without erasing historical rows.

        Validates *reason* (at least 10 non-whitespace characters), marks the
        replacing entry ``DEPRECATED`` if it is still ``ACTIVE``, restores
        *old_id* to ``ACTIVE`` via an appended row, and writes an
        ``AuditEventType.REVERT`` audit record.

        Args:
            old_id: UUID of the previously superseded rule to restore.
            reason: Human-readable justification (minimum 10 characters).

        Returns:
            The restored :class:`RuleEntry` (status ``ACTIVE``).

        Raises:
            KeyError: If *old_id* does not exist.
            ValueError: If *old_id* is not currently ``SUPERSEDED`` or *reason*
                is invalid.
        """
        validate_supersession_reason(reason)
        old_entry = self._store.get_by_id(old_id)
        if old_entry is None:
            raise KeyError(f"No rule entry found for rule_id={old_id}")
        if old_entry.status is not RuleStatus.SUPERSEDED:
            raise ValueError(
                f"Only SUPERSEDED rules can be reverted; {old_id} is {old_entry.status}"
            )
        superseding_id = old_entry.superseded_by
        if superseding_id is not None:
            new_entry = self._store.get_by_id(superseding_id)
            if new_entry is not None and new_entry.status is RuleStatus.ACTIVE:
                self._store.update_status(
                    superseding_id,
                    RuleStatus.DEPRECATED,
                    superseded_by=None,
                    reason=f"Superseded rule restored: {reason}",
                )
        self._store.update_status(
            old_id,
            RuleStatus.ACTIVE,
            superseded_by=None,
            reason=None,
        )
        self._audit.record(
            make_event(
                AuditEventType.REVERT,
                rule_id=old_id,
                detail={
                    "reverted_from_rule_id": str(superseding_id) if superseding_id else None,
                    "reason": reason,
                },
            )
        )
        restored = self._store.get_by_id(old_id)
        assert restored is not None
        return restored

    # ------------------------------------------------------------------
    # Queries
    # ------------------------------------------------------------------

    def get(self, rule_id: UUID | str) -> RuleEntry | None:
        """Return the latest state of *rule_id*, or ``None`` if not found.

        Args:
            rule_id: UUID (or string representation of a UUID) to look up.

        Returns:
            :class:`RuleEntry` or ``None``.
        """
        if isinstance(rule_id, str):
            try:
                parsed_id = UUID(rule_id)
            except ValueError:
                return None
            return self._store.get_by_id(parsed_id)
        return self._store.get_by_id(rule_id)

    def get_active(self) -> list[RuleEntry]:
        """Return all entries whose current status is ACTIVE.

        Returns:

    def register_capability(
        self,
        package: CapabilityPackage,
        evidence: VerificationEvidence | None = None,
    ) -> CapabilityPackage:
        """Register a verified capability package into the library.

        Validates verification gates (or updates verification flags using *evidence*),
        appends the package to the store with ``ACTIVE`` status, records an audit event,
        and returns the registered package.

        Args:
            package: The :class:`~evolzen.schemas.world.CapabilityPackage` to register.
            evidence: Optional :class:`~evolzen.schemas.library.VerificationEvidence`
                bundle backing the capability.

        Returns:
            The registered :class:`~evolzen.schemas.world.CapabilityPackage`.

        Raises:
            ValueError: If the package fails verification gates or if evidence validation fails.
        """
        event_type = AuditEventType.SPAWN
        if evidence is not None:
            if evidence.resource_ceiling_hit:
                raise ValueError(
                    "Verification evidence hit resource ceiling; cannot register capability"
                )
            if evidence.false_promotion_rate > 0.0:
                raise ValueError(
                    f"Verification evidence false promotion rate {evidence.false_promotion_rate} > 0.0"
                )
            ev_id = evidence.evidence_id_str
            if package.evidence_id and package.evidence_id != ev_id:
                raise ValueError(
                    f"Mismatched evidence_id between package ({package.evidence_id}) and evidence ({ev_id})"
                )
            package = package.model_copy(
                update={
                    "evidence_id": ev_id,
                    "verified_replay": True,
                    "verified_intervention": True,
                    "verified_containment": True,
                }
            )
            event_type = AuditEventType.PROMOTE
        else:
            if not (
                package.verified_replay
                and package.verified_intervention
                and package.verified_containment
            ):
                raise ValueError(
                    f"Capability package {package.package_id} failed verification gates: "
                    "verified_replay, verified_intervention, and verified_containment must all be True"
                )

        self._store.append_capability(package, status=RuleStatus.ACTIVE)
        if self._audit is not None:
            self._audit.record(
                make_event(
                    event_type,
                    rule_id=_package_id_to_uuid(package.package_id),
                    detail={
                        "package_id": package.package_id,
                        "version": package.version,
                        "action_type": str(package.action_type),
                        "evidence_id": package.evidence_id,
                    },
                )
            )
        return package

    def supersede_capability(
        self,
        old_package_id: str,
        new_package: CapabilityPackage,
        reason: str,
    ) -> CapabilityPackage:
        """Supersede an existing capability package with *new_package*.

        Validates *reason* (at least 10 non-whitespace characters), marks the
        old capability package as ``SUPERSEDED`` in the store, appends the
        new package with ``ACTIVE`` status, records an ``AuditEventType.SUPERSEDE``
        event in the audit log, and returns *new_package*.

        Args:
            old_package_id: Identifier of the package being superseded.
            new_package: Replacement :class:`~evolzen.schemas.world.CapabilityPackage`.
            reason: Human-readable justification (minimum 10 characters).

        Returns:
            The newly registered :class:`~evolzen.schemas.world.CapabilityPackage`.

        Raises:
            KeyError: If *old_package_id* does not exist in the store.
            ValueError: If *reason* is invalid or *new_package* fails verification.
        """
        validate_supersession_reason(reason)
        old = self._store.get_capability_by_id(old_package_id)
        if old is None:
            raise KeyError(f"No capability package found for package_id={old_package_id}")

        if not (
            new_package.verified_replay
            and new_package.verified_intervention
            and new_package.verified_containment
        ):
            raise ValueError(
                f"Replacement capability package {new_package.package_id} failed verification gates"
            )

        self._store.update_capability_status(old_package_id, RuleStatus.SUPERSEDED)
        self._store.append_capability(new_package, status=RuleStatus.ACTIVE)
        if self._audit is not None:
            self._audit.record(
                make_event(
                    AuditEventType.SUPERSEDE,
                    rule_id=_package_id_to_uuid(old_package_id),
                    detail={
                        "old_package_id": old_package_id,
                        "new_package_id": new_package.package_id,
                        "reason": reason,
                    },
                )
            )
        return new_package

    def register_capability(
    self,
    package: CapabilityPackage,
    evidence: VerificationEvidence | None = None,
) -> CapabilityPackage:
    """Register a verified capability package into the library.

    Validates verification gates (or updates verification flags using *evidence*),
    appends the package to the store with ``ACTIVE`` status, records an audit event,
    and returns the registered package.

    Args:
        package: The :class:`~evolzen.schemas.world.CapabilityPackage` to register.
        evidence: Optional :class:`~evolzen.schemas.library.VerificationEvidence`
            bundle backing the capability.

    Returns:
        The registered :class:`~evolzen.schemas.world.CapabilityPackage`.

    Raises:
        ValueError: If the package fails verification gates or if evidence validation fails.
    """
    event_type = AuditEventType.SPAWN
    if evidence is not None:
        if evidence.resource_ceiling_hit:
            raise ValueError(
                "Verification evidence hit resource ceiling; cannot register capability"
            )
        if evidence.false_promotion_rate > 0.0:
            raise ValueError(
                f"Verification evidence false promotion rate {evidence.false_promotion_rate} > 0.0"
            )
        ev_id = evidence.evidence_id_str
        if package.evidence_id and package.evidence_id != ev_id:
            raise ValueError(
                f"Mismatched evidence_id between package ({package.evidence_id}) and evidence ({ev_id})"
            )
        package = package.model_copy(
            update={
                "evidence_id": ev_id,
                "verified_replay": True,
                "verified_intervention": True,
                "verified_containment": True,
            }
        )
        event_type = AuditEventType.PROMOTE
    else:
        if not (
            package.verified_replay
            and package.verified_intervention
            and package.verified_containment
        ):
            raise ValueError(
                f"Capability package {package.package_id} failed verification gates: "
                "verified_replay, verified_intervention, and verified_containment must all be True"
            )

    self._store.append_capability(package, status=RuleStatus.ACTIVE)
    if self._audit is not None:
        self._audit.record(
            make_event(
                event_type,
                rule_id=_package_id_to_uuid(package.package_id),
                detail={
                    "package_id": package.package_id,
                    "version": package.version,
                    "action_type": str(package.action_type),
                    "evidence_id": package.evidence_id,
                },
            )
        )
    return package

    def revert_supersession(self, old_id: UUID, reason: str) -> RuleEntry:
    """Explicitly reverse a supersession without erasing historical rows.

    Validates *reason* (at least 10 non-whitespace characters), marks the
    replacing entry ``DEPRECATED`` if it is still ``ACTIVE``, restores
    *old_id* to ``ACTIVE`` via an appended row, and writes an
    ``AuditEventType.REVERT`` audit record.

    Args:
        old_id: UUID of the previously superseded rule to restore.
        reason: Human-readable justification (minimum 10 characters).

    Returns:
        The restored :class:`RuleEntry` (status ``ACTIVE``).

    Raises:
        KeyError: If *old_id* does not exist.
        ValueError: If *old_id* is not currently ``SUPERSEDED`` or *reason*
            is invalid.
    """
    validate_supersession_reason(reason)
    old_entry = self._store.get_by_id(old_id)
    if old_entry is None:
        raise KeyError(f"No rule entry found for rule_id={old_id}")
    if old_entry.status is not RuleStatus.SUPERSEDED:
        raise ValueError(
            f"Only SUPERSEDED rules can be reverted; {old_id} is {old_entry.status}"
        )
    superseding_id = old_entry.superseded_by
    if superseding_id is not None:
        new_entry = self._store.get_by_id(superseding_id)
        if new_entry is not None and new_entry.status is RuleStatus.ACTIVE:
            self._store.update_status(
                superseding_id,
                RuleStatus.DEPRECATED,
                superseded_by=None,
                reason=f"Superseded rule restored: {reason}",
            )
    self._store.update_status(
        old_id,
        RuleStatus.ACTIVE,
        superseded_by=None,
        reason=None,
    )
    self._audit.record(
        make_event(
            AuditEventType.REVERT,
            rule_id=old_id,
            detail={
                "reverted_from_rule_id": str(superseding_id) if superseding_id else None,
                "reason": reason,
            },
        )
    )
    restored = self._store.get_by_id(old_id)
    assert restored is not None
    return restored

    def spawn_if_novel(
    self,
    candidate: CandidateRule,
    validity: ValidityContext,
    evidence: VerificationEvidence,
    provenance: ProvenanceClass = ProvenanceClass.EXPLORATION_VERIFIED,
) -> RuleEntry | None:
    """Spawn a new library entry if *candidate* is sufficiently novel.

    Computes similarity between *candidate* (wrapped as a temporary
    :class:`RuleEntry`) and each active entry.  If the maximum similarity
    is at or above ``routing_threshold``, the candidate is *routed* to the
    existing entry and ``None`` is returned.  Otherwise a new entry is
    appended and returned.

    No existing entry is ever modified.

    Args:
        candidate: The synthesiser hypothesis to evaluate.
        validity: Validity context under which the candidate was verified.
        evidence: Verification evidence for the candidate.
        provenance: Origin classification (default:
            :attr:`~evolzen.schemas.library.ProvenanceClass.EXPLORATION_VERIFIED`).

    Returns:
        The new :class:`RuleEntry` if spawned, or ``None`` if routed.
    """
    # Build a temporary entry just for similarity scoring (not persisted).
    temp_entry = _candidate_to_entry(candidate, provenance, validity, evidence)

    active = self._store.get_active()
    best_score = 0.0
    best_match: RuleEntry | None = None
    for existing in active:
        score = self._confidence_fn(temp_entry, existing)
        if score > best_score:
            best_score = score
            best_match = existing

    if best_score >= self._threshold and best_match is not None:
        # Route — record audit event, do NOT persist temp_entry.
        self._audit.record(
            make_event(
                AuditEventType.ROUTE,
                rule_id=best_match.rule_id,
                detail={
                    "candidate_rule_id": candidate.rule_id,
                    "similarity_score": best_score,
                    "threshold": self._threshold,
                },
            )
        )
        return None

    # Spawn — persist the new entry and record audit event.
    self._store.append_rule(temp_entry)
    self._audit.record(
        make_event(
            AuditEventType.SPAWN,
            rule_id=temp_entry.rule_id,
            detail={
                "candidate_rule_id": candidate.rule_id,
                "best_similarity": best_score,
                "provenance": str(provenance),
            },
        )
    )
    return temp_entry

    class LibraryStore:
    """Append-only DuckDB store for :class:`~evolzen.schemas.library.RuleEntry` objects.

    Each public method opens its own short-lived DuckDB connection so that the
    store is safe to call from multiple processes that share the same file
    (DuckDB WAL handles concurrency).

    Args:
        db_path: Filesystem path to the ``.duckdb`` file.  Parent directories
            are created on first use.

    Example::

        store = LibraryStore(Path("data/library.duckdb"))
        store.append_rule(entry)
        retrieved = store.get_by_id(entry.rule_id)
    """

    def __init__(self, db_path: Path) -> None:
        self._db_path = db_path
        self._db_path.parent.mkdir(parents=True, exist_ok=True)
        self._ensure_schema()

    # ------------------------------------------------------------------
    # Schema bootstrap
    # ------------------------------------------------------------------

    def _ensure_schema(self) -> None:
        """Create all sequences and tables if they do not already exist.

        Raises:
            RuntimeError: If the database schema version is greater than
                :const:`SCHEMA_VERSION`.
        """
        with duckdb.connect(str(self._db_path)) as con:
            con.execute("""
                CREATE TABLE IF NOT EXISTS schema_meta (
                    schema_version INTEGER NOT NULL,
                    initialized_at TIMESTAMP NOT NULL
                )
            """)
            meta_rows = con.execute(
                "SELECT schema_version FROM schema_meta ORDER BY initialized_at DESC LIMIT 1"
            ).fetchall()
            if not meta_rows:
                con.execute(
                    "INSERT INTO schema_meta (schema_version, initialized_at) VALUES (?, current_timestamp)",
                    [SCHEMA_VERSION],
                )
            else:
                found = int(meta_rows[0][0])
                if found > SCHEMA_VERSION:
                    raise RuntimeError(
                        f"Incompatible database schema version {found} > {SCHEMA_VERSION}"
                    )

            con.execute("CREATE SEQUENCE IF NOT EXISTS rule_entries_seq START 1")
            con.execute("""
                CREATE TABLE IF NOT EXISTS rule_entries (
                    row_seq      BIGINT PRIMARY KEY DEFAULT nextval('rule_entries_seq'),
                    rule_id      TEXT NOT NULL,
                    version      INTEGER NOT NULL,
                    payload      TEXT NOT NULL
                )
            """)
            con.execute("CREATE SEQUENCE IF NOT EXISTS ss_links_seq START 1")
            con.execute("""
                CREATE TABLE IF NOT EXISTS supersession_links (
                    row_seq     BIGINT PRIMARY KEY DEFAULT nextval('ss_links_seq'),
                    old_rule_id TEXT NOT NULL,
                    new_rule_id TEXT NOT NULL,
                    payload     TEXT NOT NULL
                )
            """)
            con.execute("CREATE SEQUENCE IF NOT EXISTS reg_results_seq START 1")
            con.execute("""
                CREATE TABLE IF NOT EXISTS regression_results (
                    row_seq  BIGINT PRIMARY KEY DEFAULT nextval('reg_results_seq'),
                    check_id TEXT NOT NULL,
                    rule_id  TEXT NOT NULL,
                    payload  TEXT NOT NULL
                )
            """)
            con.execute("CREATE SEQUENCE IF NOT EXISTS cap_packages_seq START 1")
            con.execute("""
                CREATE TABLE IF NOT EXISTS capability_packages (
                    row_seq      BIGINT PRIMARY KEY DEFAULT nextval('cap_packages_seq'),
                    package_id   TEXT NOT NULL,
                    version      TEXT NOT NULL,
                    evidence_id  TEXT NOT NULL,
                    status       TEXT NOT NULL,
                    payload      TEXT NOT NULL
                )
            """)

    # Write operations
    # ------------------------------------------------------------------

    def append_rule(self, entry: RuleEntry) -> None:
        """Append a new :class:`RuleEntry` row.

        Never overwrites an existing row.  Caller is responsible for
        ensuring ``rule_id`` uniqueness (checked via
        :meth:`get_by_id` before calling).

        Args:
            entry: The rule entry to persist.
        """
        payload = entry.model_dump_json()
        with duckdb.connect(str(self._db_path)) as con:
            con.execute(
                "INSERT INTO rule_entries (rule_id, version, payload) VALUES (?, ?, ?)",
                [str(entry.rule_id), entry.version, payload],
            )

    def update_status(
        self,
        rule_id: UUID,
        status: RuleStatus,
        superseded_by: UUID | None,
        reason: str | None,
    ) -> None:
        """Record a status change by appending an updated row.

        The original row is **never** modified; this appends a new row
        carrying the updated payload so the full history is preserved.

        Args:
            rule_id: Identifier of the rule whose status is changing.
            status: New lifecycle status.
            superseded_by: If ``status`` is ``SUPERSEDED``, the replacing
                rule's id; otherwise ``None``.
            reason: Deprecation reason when ``status`` is ``DEPRECATED``.
        """
        existing = self.get_by_id(rule_id)
        if existing is None:
            raise KeyError(f"No rule entry found for rule_id={rule_id}")
        updated = existing.model_copy(
            update={
                "status": status,
                "superseded_by": superseded_by,
                "deprecation_reason": reason,
            }
        )
        payload = updated.model_dump_json()
        with duckdb.connect(str(self._db_path)) as con:
            con.execute(
                "INSERT INTO rule_entries (rule_id, version, payload) VALUES (?, ?, ?)",
                [str(rule_id), updated.version, payload],
            )

    def update_evidence(
        self,
        rule_id: UUID,
        evidence: VerificationEvidence,
    ) -> RuleEntry:
        """Record updated verification evidence by appending a new row.

        The original row is preserved intact (append-only).

        Args:
            rule_id: Identifier of the rule whose evidence is updated.
            evidence: New :class:`VerificationEvidence` bundle.

        Returns:
            The updated :class:`RuleEntry`.

        Raises:
            KeyError: If *rule_id* does not exist in the store.
        """
        existing = self.get_by_id(rule_id)
        if existing is None:
            raise KeyError(f"No rule entry found for rule_id={rule_id}")
        updated = existing.model_copy(update={"evidence": evidence})
        self.append_rule(updated)
        return updated

    def append_supersession(self, link: SupersessionLink) -> None:
        """Persist a :class:`SupersessionLink` record.

        Args:
            link: The supersession link to append.
        """
        payload = link.model_dump_json()
        with duckdb.connect(str(self._db_path)) as con:
            con.execute(
                "INSERT INTO supersession_links (old_rule_id, new_rule_id, payload) "
                "VALUES (?, ?, ?)",
                [str(link.old_rule_id), str(link.new_rule_id), payload],
            )

    def append_regression_result(self, result: RegressionResult) -> None:
        """Persist a :class:`RegressionResult` record.

        Args:
            result: The regression check outcome to store.
        """
        payload = result.model_dump_json()
        with duckdb.connect(str(self._db_path)) as con:
            con.execute(

    def get_by_id(self, rule_id: UUID) -> RuleEntry | None:
        """Return the latest state of a rule entry, or ``None`` if not found.

        "Latest" is defined by highest ``row_seq`` for the given ``rule_id``.

        Args:
            rule_id: UUID of the rule to look up.

        Returns:
            The most-recent :class:`RuleEntry` row, or ``None``.
        """
        with duckdb.connect(str(self._db_path)) as con:
            rows = con.execute(
                "SELECT payload FROM rule_entries WHERE rule_id = ? "
                "ORDER BY row_seq DESC LIMIT 1",
                [str(rule_id)],
            ).fetchall()
        if not rows:
            return None
        return RuleEntry.model_validate_json(rows[0][0])

    def get_all(self) -> list[RuleEntry]:
        """Return the latest state of every rule entry in the store.

        Uses a window function to deduplicate: only the row with the highest
        ``row_seq`` per ``rule_id`` is returned.

        Returns:
            List of latest :class:`RuleEntry` objects (all statuses).
        """
        with duckdb.connect(str(self._db_path)) as con:
            rows = con.execute("""
                SELECT payload
                FROM (
                    SELECT row_seq, payload, ROW_NUMBER() OVER (
                        PARTITION BY rule_id ORDER BY row_seq DESC
                    ) AS rn
                    FROM rule_entries
                ) sub
                WHERE rn = 1
                ORDER BY row_seq
            """).fetchall()
        return [RuleEntry.model_validate_json(row[0]) for row in rows]

    def get_active(self) -> list[RuleEntry]:
        """Return only entries whose current status is ``ACTIVE``.

        Returns:
            List of active :class:`RuleEntry` objects.
        """
        return [e for e in self.get_all() if e.status == RuleStatus.ACTIVE]

    def get_family(self, rule_family_id: UUID) -> list[RuleEntry]:
        """Return all versions sharing the given ``rule_family_id``.

        Returns one entry per ``rule_id`` (latest state of each), ordered by
        ascending ``version``.

        Args:
            rule_family_id: Family UUID to filter by.

        Returns:
            Ordered list of :class:`RuleEntry` objects in the family.
        """
        all_entries = self.get_all()
        family = [e for e in all_entries if e.rule_family_id == rule_family_id]
        return sorted(family, key=lambda e: e.version)

    def get_supersession_links(
        self, rule_id: UUID | None = None
    ) -> list[SupersessionLink]:
        """Return persisted :class:`SupersessionLink` records in insertion order.

        Args:
            rule_id: If provided, filter to links where *rule_id* appears as
                either ``old_rule_id`` or ``new_rule_id``.

        Returns:
            List of :class:`SupersessionLink` objects.
        """
        with duckdb.connect(str(self._db_path)) as con:
            if rule_id is None:
                rows = con.execute(
                    "SELECT payload FROM supersession_links ORDER BY row_seq"
                ).fetchall()
            else:
                rid = str(rule_id)
                rows = con.execute(
                    "SELECT payload FROM supersession_links "
                    "WHERE old_rule_id = ? OR new_rule_id = ? ORDER BY row_seq",
                    [rid, rid],
                ).fetchall()
        return [SupersessionLink.model_validate_json(row[0]) for row in rows]

    def count(self) -> int:
        """Total number of distinct rule IDs in the store (all statuses).

        Returns:
            Integer count of distinct ``rule_id`` values.
        """
        with duckdb.connect(str(self._db_path)) as con:
            result = con.execute(
                "SELECT COUNT(DISTINCT rule_id) FROM rule_entries"
            ).fetchone()
        return int(result[0]) if result else 0

    def count_active(self) -> int:
        """Number of rules whose current status is ``ACTIVE``.

        Returns:
            Integer count of active rule IDs.
        """
        return len(self.get_active())

    class AuditLog:
    """Append-only JSONL audit log.

    One JSON line per :class:`~evolzen.schemas.library.AuditEvent`.  Lines are
    never removed or rewritten.  The parent directory is created on first use.

    Args:
        log_path: Path to the ``.jsonl`` file.

    Example::

        log = AuditLog(Path("data/library_audit.jsonl"))
        event = make_event(AuditEventType.SPAWN, rule_id=some_uuid)
        log.record(event)
    """

    def __init__(self, log_path: Path) -> None:
        self._log_path = log_path
        self._log_path.parent.mkdir(parents=True, exist_ok=True)

    def record(self, event: AuditEvent) -> None:
        """Append *event* as a single JSON line.

        The write is performed under an exclusive file lock to prevent
        interleaved writes from concurrent processes.

        Args:
            event: The :class:`~evolzen.schemas.library.AuditEvent` to persist.
        """
        line = event.model_dump_json() + "\n"
        with self._log_path.open("a", encoding="utf-8") as fh:
            fh.seek(0)
            _lock_file(fh)
            try:
                fh.write(line)
                fh.flush()
            finally:
                fh.seek(0)
                _unlock_file(fh)

    def load_all(self) -> list[AuditEvent]:
        """Return all events in the log, in insertion order.

        Returns:
            List of :class:`~evolzen.schemas.library.AuditEvent` objects.
            Returns an empty list if the log file does not exist or is empty.
        """
        if not self._log_path.exists():
            return []
        events: list[AuditEvent] = []
        for raw_line in self._log_path.read_text(encoding="utf-8").splitlines():
            line = raw_line.strip()
            if line:
                events.append(AuditEvent.model_validate_json(line))
        return events

        Returns:
            Filtered list of :class:`~evolzen.schemas.library.AuditEvent` objects.
        """

        """Return the total number of events recorded in the log.

        Returns:

        return len(self.load_all())

    class RegressionBank:
    """DuckDB-backed store for held-out regression fixtures and check results.

    Fixtures are **append-only**: once registered they are never modified or
    deleted.  This guarantees the held-out set only grows, so the zero-forgetting
    metric can be measured at any point in time.

    Args:
        db_path: Filesystem path to the ``.duckdb`` file (may be shared with
            :class:`~evolzen.library.persistence.LibraryStore` or separate).

    Example::

        bank = RegressionBank(Path("data/library.duckdb"))
        bank.register(rule_id, [tid1, tid2], [fixture1, fixture2])
        result = bank.check_rule(rule_id, entry, store)
    """

    def __init__(self, db_path: Path) -> None:
        self._db_path = db_path
        self._db_path.parent.mkdir(parents=True, exist_ok=True)
        self._ensure_schema()

    # ------------------------------------------------------------------
    # Schema bootstrap
    # ------------------------------------------------------------------

    def _ensure_schema(self) -> None:
        """Create regression tables and sequences if they do not exist."""
        with duckdb.connect(str(self._db_path)) as con:
            con.execute("CREATE SEQUENCE IF NOT EXISTS reg_fixtures_seq START 1")
            con.execute("""
                CREATE TABLE IF NOT EXISTS regression_fixtures (
                    row_seq        BIGINT PRIMARY KEY DEFAULT nextval('reg_fixtures_seq'),
                    rule_id        TEXT NOT NULL,
                    transition_id  TEXT NOT NULL,
                    fixture_json   TEXT NOT NULL
                )
            """)
            con.execute("CREATE SEQUENCE IF NOT EXISTS reg_results_seq START 1")
            con.execute("""
                CREATE TABLE IF NOT EXISTS regression_results (
                    row_seq  BIGINT PRIMARY KEY DEFAULT nextval('reg_results_seq'),
                    check_id TEXT NOT NULL,
                    rule_id  TEXT NOT NULL,
                    payload  TEXT NOT NULL
                )
            """)

    # ------------------------------------------------------------------
    # Registration
    # ------------------------------------------------------------------

    def register(
        self,
        rule_id: UUID,
        transition_ids: Sequence[UUID],
        fixtures: Sequence[dict[str, Any]],
    ) -> None:
        """Store held-out fixtures for *rule_id*.

        Each fixture is a plain serialisable dict containing at minimum:
        - ``"transition_id"``: matches the corresponding entry in *transition_ids*.
        - ``"expected_emission"``: the emission the rule is expected to predict.
        - Any additional context fields needed for the check.

        Existing fixtures are never removed (append-only).

        Args:
            rule_id: UUID of the rule these fixtures belong to.
            transition_ids: Sequence of transition UUIDs (one per fixture).
            fixtures: Sequence of fixture dicts (must be JSON-serialisable).

        Raises:
            ValueError: If *transition_ids* and *fixtures* have different lengths.
        """
        if len(transition_ids) != len(fixtures):
            raise ValueError(
                f"transition_ids ({len(transition_ids)}) and fixtures ({len(fixtures)}) "
                "must have the same length"
            )
        with duckdb.connect(str(self._db_path)) as con:
            for tid, fixture in zip(transition_ids, fixtures, strict=True):
                con.execute(
                    "INSERT INTO regression_fixtures (rule_id, transition_id, fixture_json) "
                    "VALUES (?, ?, ?)",
                    [str(rule_id), str(tid), json.dumps(fixture)],
                )

    def register_fixtures(
        self,
        rule_id: UUID,
        transition_ids: Sequence[UUID],
        fixtures: Sequence[dict[str, Any]],
    ) -> None:
        """Store held-out fixtures for *rule_id*.

        Alias for :meth:`register` to support explicit fixture registration.

        Args:
            rule_id: UUID of the rule these fixtures belong to.
            transition_ids: Sequence of transition UUIDs (one per fixture).
            fixtures: Sequence of fixture dicts (must be JSON-serialisable).
        """
        self.register(rule_id, transition_ids, fixtures)

    def register_from_transitions(
        self,
        rule_id: UUID,
        transitions: Sequence[Transition],
        expected_emission_fn: Callable[[Transition], dict[str, Any]] | None = None,
    ) -> list[UUID]:
        """Convert recorded transitions into regression fixtures and register them.

        Security Note: Fixtures store only plain serialisable dictionaries and
        never execute arbitrary rule code.

        Args:
            rule_id: UUID of the rule these fixtures validate.
            transitions: Sequence of recorded :class:`~evolzen.schemas.transition.Transition`
                records from the environment.
            expected_emission_fn: Optional callable to extract expected emission
                dict from a transition. Defaults to `transition.emission.data`.

        Returns:
            List of transition UUIDs registered as fixtures.
        """
        transition_ids: list[UUID] = []
        fixtures: list[dict[str, Any]] = []

        for t in transitions:
            tid = t.transition_id
            expected = (
                expected_emission_fn(t)
                if expected_emission_fn is not None
                else dict(t.emission.data)
            )
            fixture: dict[str, Any] = {
                "transition_id": str(tid),
                "expected_emission": expected,
                "action_type": t.action.action_type,
                "agent_id": t.action.agent_id,
                "state_id": str(t.state.state_id),
                "next_state_id": str(t.next_state.state_id),
                "seed": t.seed,
            }
            transition_ids.append(tid)
            fixtures.append(fixture)

        self.register_fixtures(rule_id, transition_ids, fixtures)
        return transition_ids

    def register_from_transition_store(
        self,
        rule_id: UUID,
        store: TransitionLogStore,
        run_id: str,
        up_to_tick: int | None = None,
        expected_emission_fn: Callable[[Transition], dict[str, Any]] | None = None,
    ) -> list[UUID]:
        """Load transitions from a `TransitionLogStore` and register them as fixtures.

        Args:
            rule_id: UUID of the rule to register fixtures for.
            store: The :class:`~evolzen.storage.transition_log.TransitionLogStore`
                instance to load transitions from.
            run_id: Run identifier to filter transitions by.
            up_to_tick: Optional maximum tick (inclusive) to load up to.
            expected_emission_fn: Optional callable to extract expected emission
                dict from a transition. Defaults to `transition.emission.data`.

        Returns:
            List of transition UUIDs registered as fixtures.
        """
        transitions = store.list_transitions(run_id=run_id, to_tick=up_to_tick)
        return self.register_from_transitions(
            rule_id=rule_id,
            transitions=transitions,
            expected_emission_fn=expected_emission_fn,
        )

    def check_rule(
        self,
        rule_id: UUID,
        entry: RuleEntry,  # noqa: ARG002
        store: Any = None,  # kept for API symmetry; not used in fixture-only check
        audit_log: AuditLog | None = None,
    ) -> RegressionResult:
        """Run the held-out fixtures for *rule_id* and store the result.

        SECURITY NOTE: This method does **not** execute ``entry.code`` or any
        synthesiser output.  It compares the ``"expected_emission"`` field in
        each registered fixture against the ``"actual_emission"`` field (if
        present) or falls back to treating the fixture as passing when
        ``"expected_emission"`` equals ``"actual_emission"`` both set to the
        same value.  All fixture data is pre-verified serialisable dicts.
        # noqa: S307

        A fixture *passes* when ``fixture["expected_emission"]`` equals
        ``fixture.get("actual_emission", fixture["expected_emission"])``.
        This convention lets callers register fixtures that pre-compute
        ground truth without coupling to any executable rule code.

        Args:
            rule_id: UUID of the rule to check.
            entry: The :class:`~evolzen.schemas.library.RuleEntry` (used for
                schema completeness; code is not executed here).
            store: Optional store reference (unused; kept for API symmetry with
                :meth:`check_all_active`).

        Returns:
            A :class:`~evolzen.schemas.library.RegressionResult` (also persisted).
        """
        fixtures = self.cases_for_rule(rule_id)
        failed: list[UUID] = []
        checked_ids: list[UUID] = []

        for fixture in fixtures:
            tid = UUID(fixture.get("transition_id", str(uuid4())))
            checked_ids.append(tid)
            expected = fixture.get("expected_emission")
            actual = fixture.get("actual_emission", expected)
            if expected != actual:
                failed.append(tid)

        result = RegressionResult(
            check_id=uuid4(),
            rule_id=rule_id,
            transition_ids_checked=tuple(checked_ids),
            all_passed=len(failed) == 0,
            failed_transition_ids=tuple(failed),
            checked_at=datetime.now(tz=UTC),
        )
        self._persist_result(result)
        if audit_log is not None:
            audit_log.record(
                make_event(
                    AuditEventType.REGRESSION_CHECK,
                    rule_id=rule_id,
                    detail={
                        "check_id": str(result.check_id),
                        "all_passed": result.all_passed,
                        "checked_count": len(checked_ids),
                        "failed_count": len(failed),
                    },
                )
            )
        return result

    def check_rule(
    self,
    rule_id: UUID,
    entry: RuleEntry,  # noqa: ARG002
    store: Any = None,  # kept for API symmetry; not used in fixture-only check
    audit_log: AuditLog | None = None,
) -> RegressionResult:
    """Run the held-out fixtures for *rule_id* and store the result.

    SECURITY NOTE: This method does **not** execute ``entry.code`` or any
    synthesiser output.  It compares the ``"expected_emission"`` field in
    each registered fixture against the ``"actual_emission"`` field (if
    present) or falls back to treating the fixture as passing when
    ``"expected_emission"`` equals ``"actual_emission"`` both set to the
    same value.  All fixture data is pre-verified serialisable dicts.
    # noqa: S307

    A fixture *passes* when ``fixture["expected_emission"]`` equals
    ``fixture.get("actual_emission", fixture["expected_emission"])``.
    This convention lets callers register fixtures that pre-compute
    ground truth without coupling to any executable rule code.

    Args:
        rule_id: UUID of the rule to check.
        entry: The :class:`~evolzen.schemas.library.RuleEntry` (used for
            schema completeness; code is not executed here).
        store: Optional store reference (unused; kept for API symmetry with
            :meth:`check_all_active`).

    Returns:
        A :class:`~evolzen.schemas.library.RegressionResult` (also persisted).
    """
    fixtures = self.cases_for_rule(rule_id)
    failed: list[UUID] = []
    checked_ids: list[UUID] = []

    for fixture in fixtures:
        tid = UUID(fixture.get("transition_id", str(uuid4())))
        checked_ids.append(tid)
        expected = fixture.get("expected_emission")
        actual = fixture.get("actual_emission", expected)
        if expected != actual:
            failed.append(tid)

    result = RegressionResult(
        check_id=uuid4(),
        rule_id=rule_id,
        transition_ids_checked=tuple(checked_ids),
        all_passed=len(failed) == 0,
        failed_transition_ids=tuple(failed),
        checked_at=datetime.now(tz=UTC),
    )
    self._persist_result(result)
    if audit_log is not None:
        audit_log.record(
            make_event(
                AuditEventType.REGRESSION_CHECK,
                rule_id=rule_id,
                detail={
                    "check_id": str(result.check_id),
                    "all_passed": result.all_passed,
                    "checked_count": len(checked_ids),
                    "failed_count": len(failed),
                },
            )
        )
    return result

    def register_from_transitions(
    self,
    rule_id: UUID,
    transitions: Sequence[Transition],
    expected_emission_fn: Callable[[Transition], dict[str, Any]] | None = None,
) -> list[UUID]:
    """Convert recorded transitions into regression fixtures and register them.

    Security Note: Fixtures store only plain serialisable dictionaries and
    never execute arbitrary rule code.

    Args:
        rule_id: UUID of the rule these fixtures validate.
        transitions: Sequence of recorded :class:`~evolzen.schemas.transition.Transition`
            records from the environment.

            dict from a transition. Defaults to `transition.emission.data`.

    Returns:
        List of transition UUIDs registered as fixtures.

        tid = t.transition_id
        expected = (
            expected_emission_fn(t)
            if expected_emission_fn is not None
            else dict(t.emission.data)
        )
        fixture: dict[str, Any] = {
            "transition_id": str(tid),
            "expected_emission": expected,
            "action_type": t.action.action_type,
            "agent_id": t.action.agent_id,
            "state_id": str(t.state.state_id),
            "next_state_id": str(t.next_state.state_id),
            "seed": t.seed,
        }
        transition_ids.append(tid)
        fixtures.append(fixture)

    self.register_fixtures(rule_id, transition_ids, fixtures)
    return transition_ids
